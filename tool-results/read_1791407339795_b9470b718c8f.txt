    47→-- ============================================================
    48→-- ATTACHMENT REGISTRATION
    49→-- ============================================================
    50→-- Find every attachment file ONCE at addon-load time and:
    51→--   * SERVER: AddCSLuaFile() it so it gets pushed to every client
    52→--     that ever connects. Doing this during addon load (rather than
    53→--     inside a Think hook) guarantees clients receive the file BEFORE
    54→--     they try to include() it. The previous code called AddCSLuaFile
    55→--     inside RegisterAttachments (a Think hook), which was too late
    56→--     for already-connected clients and caused include() to fail
    57→--     silently under pcall — so CustomUH.Attachments stayed empty
    58→--     and the menu showed slot headers with zero attachment buttons.
    59→--   * BOTH: include() it and register the ATTACHMENT table.
    60→-- ============================================================
    61→
    62→CustomUH._attFiles = file.Find("cuh_attachments/*.lua", "LUA")
    63→
    64→if SERVER then
    65→    for _, filename in ipairs(CustomUH._attFiles) do
    66→        AddCSLuaFile("cuh_attachments/" .. filename)
    67→    end

    70→function CustomUH.RegisterAttachments()
    71→    -- Use the file list captured at addon-load time. On a server-only
    72→    -- install (where the client never had the addon mounted locally),
    73→    -- the server pushes this list over net via CUH2_AttList and the
    74→    -- client replaces CustomUH._attFiles with the networked copy.
    75→    local files = CustomUH._attFiles or {}
    76→    print("[CUH-DBG] RegisterAttachments CALLED  realm=" .. (SERVER and "SERVER" or "CLIENT")
    77→        .. "  file count=" .. #files)
    78→    for i, fn in ipairs(files) do
    79→        print("[CUH-DBG]   file[" .. i .. "] = " .. tostring(fn))
    80→    end
    81→    for _, filename in ipairs(files) do
    82→        ATTACHMENT = {}
    83→        local path = "cuh_attachments/" .. filename
    84→
    85→        local ok, err = pcall(include, path)
    86→        if not ok then
    87→            print("[CUH-DBG]   Error loading " .. filename .. ": " .. tostring(err))
    88→            print("[CustomUH] Error loading " .. filename .. ": " .. tostring(err))
    89→        end
    90→
    91→        if ATTACHMENT and ATTACHMENT.Name then
    92→            local id = string.StripExtension(filename)
    93→            ATTACHMENT.ID = id
    94→            CustomUH.Attachments[id] = ATTACHMENT
    95→            print("[CUH-DBG]   registered attachment id='" .. id .. "'  Name='" .. tostring(ATTACHMENT.Name) .. "'  PartClass=" .. tostring(ATTACHMENT.PartClass))
    96→        else
    97→            print("[CUH-DBG]   ATTACHMENT skipped (no .Name): " .. filename)
    98→            print("[CustomUH] Attachment skipped (no .Name): " .. filename)
    99→        end
   100→        ATTACHMENT = nil
   101→    end
   102→    print("[CUH-DBG] RegisterAttachments DONE  total registered = " .. table.Count(CustomUH.Attachments))
   103→end

   132→function CustomUH.LoadAttachments(wep, ply)
   133→    if not IsValid(ply) then ply = wep:GetOwner() end
   134→    if not IsValid(ply) then return end
   135→    if not wep.Attachments then return end
   136→
   137→    local path = CustomUH.GetSavePath(ply)
   138→    if not path or not file.Exists(path, "DATA") then
   139→        print("[CUH-DBG] LoadAttachments: no save file at " .. tostring(path))
   140→        return
   141→    end
   142→
   143→    local data = util.JSONToTable(file.Read(path, "DATA") or "")
   144→    if not data then
   145→        print("[CUH-DBG] LoadAttachments: save file is invalid JSON")
   146→        return
   147→    end
   148→
   149→    local class = wep:GetClass()
   150→    local saved = data[class]
   151→    if not saved then
   152→        print("[CUH-DBG] LoadAttachments: no save entry for " .. class)
   153→        return
   154→    end
   155→
   156→    print("[CUH-DBG] LoadAttachments: loading saved selections for " .. class)
   157→    local changed = false
   158→    local tempData = {}
   159→
   160→    for slot, index in pairs(saved) do
   161→        slot = tonumber(slot) or slot
   162→        if wep.Attachments[slot] then
   163→            if index == 0 or (wep.Attachments[slot].atts and index <= #wep.Attachments[slot].atts) then
   164→                wep.Attachments[slot].sel = index
   165→                changed = true
   166→                tempData[#tempData + 1] = { slot = slot, index = index }
   167→                print("[CUH-DBG]   loaded slot=" .. slot .. "  index=" .. index)
   168→            end
   169→        end
   170→    end
   171→
   172→    if changed then
   173→        wep:ApplyAttachments()
   174→
   175→        -- Send saved state to client
   176→        net.Start("CUH2_AttLoad")
   177→            net.WriteEntity(wep)
   178→            net.WriteUInt(#tempData, 8)
   179→            for i = 1, #tempData do
   180→                net.WriteUInt(tempData[i].slot, 8)
   181→                net.WriteUInt(tempData[i].index, 8)
   182→            end
   183→        net.Send(ply)
   184→    end
   185→end

   187→function CustomUH.SaveAttachments(wep, ply)
   188→    if not IsValid(ply) then ply = wep:GetOwner() end
   189→    if not IsValid(ply) then return end
   190→    if not wep.Attachments then return end
   191→
   192→    local path = CustomUH.GetSavePath(ply)
   193→    if not path then return end
   194→
   195→    local data = {}
   196→    if file.Exists(path, "DATA") then
   197→        data = util.JSONToTable(file.Read(path, "DATA") or "") or {}
   198→    end
   199→
   200→    local class = wep:GetClass()
   201→    local current = {}
   202→    local hasAttachments = false
   203→
   204→    for slot, slotData in pairs(wep.Attachments) do
   205→        if slotData.sel and slotData.sel > 0 then
   206→            current[tostring(slot)] = slotData.sel
   207→            hasAttachments = true
   208→        else
   209→            current[tostring(slot)] = 0
   210→        end
   211→    end
   212→
   213→    if hasAttachments then
   214→        data[class] = current
   215→    else
   216→        data[class] = nil
   217→    end
   218→
   219→    if not file.Exists(SAVE_DIR, "DATA") then
   220→        file.CreateDir(SAVE_DIR)
   221→    end
   222→
   223→    file.Write(path, util.TableToJSON(data, true))
   224→    print("[CUH-DBG] SaveAttachments: saved to " .. path .. "  class=" .. class .. "  entries=" .. table.Count(current))
   225→end

   230→        if wep.Attachments then
   231→            CustomUH.SaveAttachments(wep, ply)
   232→        end
   233→    end

   246→-- ============================================================
   247→-- CUH2_AttList — server pushes the attachment filename list to
   248→-- clients that don't have the addon mounted locally. The client
   249→-- then include()s each (which works because we AddCSLuaFile'd
   250→-- them at addon-load time on the server).
   251→-- ============================================================
   252→if SERVER then
   253→    local function SendAttList(ply)
   254→        net.Start("CUH2_AttList")
   255→            net.WriteUInt(#CustomUH._attFiles, 8)
   256→            for _, fn in ipairs(CustomUH._attFiles) do
   257→                net.WriteString(fn)
   258→            end
   259→        net.Send(ply)
   260→    end
   261→
   262→    -- Send to a player as soon as they finish initializing
   263→    hook.Add("PlayerInitialSpawn", "CUH2_SendAttList", function(ply)
   264→        timer.Simple(1, function()
   265→            if IsValid(ply) then SendAttList(ply) end
   266→        end)
   267→    end)
   268→end
   269→
   270→if CLIENT then
   271→    net.Receive("CUH2_AttList", function()
   272→        local n = net.ReadUInt(8)
   273→        local list = {}
   274→        for i = 1, n do
   275→            list[i] = net.ReadString()
   276→        end
   277→        -- Only adopt the server's list if we didn't find any files
   278→        -- locally — preserves the local file.Find result when the
   279→        -- addon is mounted on the client (which is faster and works
   280→        -- even before PlayerInitialSpawn fires).
   281→        if not CustomUH._attFiles or #CustomUH._attFiles == 0 then
   282→            CustomUH._attFiles = list
   283→            CustomUH.RegisterAttachments()
   284→        end
   285→    end)
   286→end

   288→if SERVER then
   289→    net.Receive("CUH2_AttSelect", function(len, ply)
   290→        local wep = net.ReadEntity()
   291→        local slot = net.ReadUInt(8)
   292→        local index = net.ReadUInt(8)
   293→
   294→        print("[CUH-DBG] SERVER net CUH2_AttSelect  ply=" .. tostring(ply) .. "  wep=" .. tostring(wep)
   295→            .. "  slot=" .. tostring(slot) .. "  index=" .. tostring(index))
   296→
   297→        if not IsValid(wep) then
   298→            print("[CUH-DBG]   ABORT: wep not valid")
   299→            return
   300→        end
   301→        if wep:GetOwner() ~= ply then
   302→            print("[CUH-DBG]   ABORT: owner mismatch  owner=" .. tostring(wep:GetOwner()))
   303→            return
   304→        end
   305→        if not wep.Attachments or not wep.Attachments[slot] then
   306→            print("[CUH-DBG]   ABORT: wep.Attachments or slot missing")
   307→            return
   308→        end
   309→
   310→        -- Validate index
   311→        if index > 0 and (not wep.Attachments[slot].atts or index > #wep.Attachments[slot].atts) then
   312→            print("[CUH-DBG]   ABORT: index out of range")
   313→            return
   314→        end
   315→
   316→        -- forceDefault check
   317→        if index == 0 and wep.Attachments[slot].forceDefault
   318→            and wep.Attachments[slot].default and wep.Attachments[slot].default > 0 then
   319→            index = wep.Attachments[slot].default
   320→            print("[CUH-DBG]   forceDefault snapped 0 -> " .. tostring(index))
   321→        end
   322→
   323→        wep.Attachments[slot].sel = index
   324→        print("[CUH-DBG]   SERVER calling wep:ApplyAttachments() ...")
   325→        wep:ApplyAttachments()
   326→        print("[CUH-DBG]   SERVER calling CustomUH.SaveAttachments ...")
   327→        CustomUH.SaveAttachments(wep, ply)
   328→
   329→        -- Broadcast to all clients
   330→        net.Start("CUH2_AttSync")
   331→            net.WriteEntity(wep)
   332→            net.WriteUInt(slot, 8)
   333→            net.WriteUInt(index, 8)
   334→        net.Broadcast()
   335→        print("[CUH-DBG]   SERVER broadcast CUH2_AttSync done")
   336→    end)
   337→end
   338→
   339→if CLIENT then
   340→    net.Receive("CUH2_AttSync", function()
   341→        local wep = net.ReadEntity()
   342→        local slot = net.ReadUInt(8)
   343→        local index = net.ReadUInt(8)
   344→
   345→        print("[CUH-DBG] CLIENT net CUH2_AttSync  wep=" .. tostring(wep) .. "  slot=" .. tostring(slot) .. "  index=" .. tostring(index))
   346→
   347→        if IsValid(wep) and wep.Attachments and wep.Attachments[slot] then
   348→            wep.Attachments[slot].sel = index
   349→            print("[CUH-DBG]   CLIENT calling wep:ApplyAttachments() ...")

   359→
   360→        if not IsValid(wep) or not wep.Attachments then return end
   361→
   362→        for i = 1, count do
   363→            local slot = net.ReadUInt(8)
   364→            local index = net.ReadUInt(8)
   365→            if wep.Attachments[slot] then
   366→                wep.Attachments[slot].sel = index
   367→            end
   368→        end
   369→        wep:ApplyAttachments()
   370→    end)

   374→-- HOOKS — auto save/load
   375→-- ============================================================
   376→
   377→-- Load on weapon switch — PlayerSwitchWeapon is a real GMod hook
   378→-- that fires when a player switches TO a new weapon.
   379→-- The old code used "WeaponDeployed" which is NOT a standard GMod
   380→-- hook and was never called, so LoadAttachments never ran and saved
   381→-- attachments were never loaded back.
   382→hook.Add("PlayerSwitchWeapon", "CUH2_LoadOnSwitch", function(ply, oldWep, newWep)
   383→    if not IsValid(ply) or not IsValid(newWep) then return end
   384→    if not newWep.IsCUHWeapon then return end
   385→    if not newWep.Attachments then return end
   386→    if SERVER then
   387→        -- Defer to next tick so the weapon is fully initialized
   388→        timer.Simple(0, function()
   389→            if IsValid(newWep) and IsValid(ply) then
   390→                CustomUH.LoadAttachments(newWep, ply)
   391→            end
   392→        end)
   393→    end
   394→end)
   395→
   396→-- Save on player death
   397→hook.Add("PlayerDeath", "CUH2_SaveOnDeath", function(ply)
   398→    if IsValid(ply) then CustomUH.SaveAllAttachments(ply) end
   399→end)
   400→
   401→-- Save on disconnect
   402→hook.Add("PlayerDisconnected", "CUH2_SaveOnDisconnect", function(ply)
   403→    if IsValid(ply) then CustomUH.SaveAllAttachments(ply) end
   404→end)

   520→-- cuh_apply — manually triggers ApplyAttachments on the active weapon.

   524→    local wep = ply:GetActiveWeapon()
