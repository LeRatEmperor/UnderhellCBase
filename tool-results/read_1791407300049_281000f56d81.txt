   169→function SWEP:EasySendWeaponAnim(lookupanimation, elseifnotfounded)
   170→    local vm = IsValid(self.Owner) and self.Owner:GetViewModel() or nil
   171→    if not IsValid(vm) then return end
   172→
   173→    if self.Animations and self.Animations[lookupanimation] then
   174→        local animData = self.Animations[lookupanimation]
   175→        if istable(animData) then
   176→            animData = animData[math.random(1, #animData)]
   177→        end
   178→        local seq = vm:LookupSequence(animData)
   179→        if seq and seq >= 0 then  -- FIXED: was > 0
   180→            self:SendSequence(vm, seq)
   181→        else
   182→            local act = type(elseifnotfounded) == "number" and vm:SelectWeightedSequence(elseifnotfounded) or -1
   183→            if act >= 0 then self:SendSequence(vm, act) end
   184→        end
   185→    else
   186→        local act = type(elseifnotfounded) == "number" and vm:SelectWeightedSequence(elseifnotfounded) or -1
   187→        if act >= 0 then self:SendSequence(vm, act) end
   188→    end
   189→    vm:SetCycle(0)
   190→    self:SetupAnimSounds(lookupanimation)
   191→end

   212→-- ============================================================
   213→-- AnimSounds system
   214→-- ============================================================
   215→function SWEP:SetupAnimSounds(animKey, variantName)
   216→    if not self.AnimSounds then return end
   217→    local timeline = nil
   218→    if variantName and self.AnimSounds[variantName] then
   219→        timeline = self.AnimSounds[variantName]
   220→    end
   221→    if not istable(timeline) or #timeline == 0 then
   222→        timeline = self.AnimSounds[animKey]
   223→    end
   224→    if (not timeline or not istable(timeline)) and animKey == "reload_empty" then
   225→        timeline = self.AnimSounds["reload"]
   226→    end
   227→    if not istable(timeline) or #timeline == 0 then return end
   228→
   229→    self._animSoundTimeline = timeline
   230→    self._animSoundIndex = 1
   231→    self._animSoundStartTime = CurTime()
   232→    local vm = IsValid(self.Owner) and self.Owner:GetViewModel() or nil
   233→    self._animSoundPlaybackRate = IsValid(vm) and vm:GetPlaybackRate() or 1
   234→    self._animSoundActive = true
   235→end
   236→
   237→function SWEP:ProcessAnimSounds()
   238→    if not self._animSoundActive then return end
   239→    local sp = game.SinglePlayer()
   240→    local iftp = IsFirstTimePredicted()
   241→    -- FIXED: standardized prediction gate — server always runs for MP broadcast
   242→    if not (SERVER or (CLIENT and iftp)) then return end
   243→
   244→    local owner = self:GetOwner()
   245→    if not IsValid(owner) then
   246→        self._animSoundActive = false
   247→        return
   248→    end
   249→
   250→    local timeline = self._animSoundTimeline
   251→    local idx = self._animSoundIndex
   252→    if not timeline or idx > #timeline then
   253→        self._animSoundActive = false
   254→        return
   255→    end
   256→
   257→    local elapsed = (CurTime() - self._animSoundStartTime) * (self._animSoundPlaybackRate or 1)
   258→    while idx <= #timeline and elapsed >= (timeline[idx].time or 0) do
   259→        local entry = timeline[idx]
   260→        if entry.sound and entry.sound ~= "" and entry.sound ~= "nil" then
   261→            local snd = entry.sound
   262→            if istable(snd) then snd = snd[math.random(1, #snd)] end
   263→            owner:EmitSound(snd, entry.level or 75, entry.pitch or 100, 1, CHAN_USER_BASE)
   264→        end
   265→        if entry.callback then entry.callback(self) end
   266→        idx = idx + 1
   267→    end
   268→    self._animSoundIndex = idx
   269→end
   270→
   271→function SWEP:ClearAnimSounds()
   272→    self._animSoundActive = false
   273→    self._animSoundTimeline = nil
   274→    self._animSoundIndex = 0
   275→end

   316→-- PrimaryAttack
   317→-- ============================================================
   318→function SWEP:PrimaryAttack()
   319→    if not self:CanPrimaryAttack() then return end
   320→    local sp = game.SinglePlayer()
   321→    local iftp = IsFirstTimePredicted()
   322→
   323→    -- Firemode-specific shoot behavior
   324→    if SERVER or iftp then
   325→        local mode = self.FireModes[self:GetNWInt("FireMode")]
   326→        if mode and mode.shoot then
   327→            local res = mode.shoot(self.Owner, self)
   328→            if res then return end
   329→        end
   330→
   331→        -- FIXED: local spread (was global)
   332→        local dmg = math.random(self.Primary.MinDamage, self.Primary.MaxDamage)
   333→        if self:GetNWBool("Silenced") then
   334→            dmg = math.Round(dmg * 0.95)
   335→        end
   336→        self:ShootBullets(self.Owner:GetShootPos(), self.Owner:GetAimVector(), dmg, self.Penetration or 2)
   337→    end
   338→
   339→    local recoil = util.SharedRandom("uh_recoil", self.Primary.MinRecoil, self.Primary.MaxRecoil) * (self:GetUHBool("Zooming") and 0.35 or 1)
   340→
   341→    -- FIXED: standardized prediction gate — SERVER runs for MP broadcast
   342→    if SERVER or (CLIENT and iftp) then
   343→        self:DoMuzzleFlash()
   344→        self:CreateSmoke(self:GetMuzzle(), self.Primary.Delay + (self.Primary.Automatic and 0.14 or 0.32))
   345→        if not self.NoShell then
   346→            self:CreateShell(self.ShellDelay or 0, self.ShellHeat)
   347→        end
   348→        self.Owner:SetEyeAngles(self.Owner:EyeAngles() + Angle(recoil, 0, 0))
   349→    end
   350→
   351→    self.Owner:ViewPunch(Angle(recoil, 0, 0))
   352→
   353→    -- Animation
   354→    self:SendWeaponAnim(ACT_VM_IDLE)
   355→    local baseAct = ACT_VM_PRIMARYATTACK
   356→    if self.ShootAnimation then
   357→        baseAct = self:ShootAnimation()
   358→    end
   359→    local isSilenced = self:GetNWBool("Silenced")
   360→    local isZooming = self:GetUHBool("Zooming")
   361→    local lookupKey = isZooming and "shoot_ads" or "shoot"
   362→    if isSilenced then lookupKey = lookupKey .. "_sil" end
   363→    if not self.Animations or not self.Animations[lookupKey] then
   364→        lookupKey = tostring(baseAct)
   365→        if isSilenced then lookupKey = lookupKey .. "_sil" end
   366→    end
   367→
   368→    local fallbackActivity = baseAct
   369→    if isSilenced then
   370→        if baseAct == ACT_VM_PRIMARYATTACK then
   371→            fallbackActivity = ACT_VM_PRIMARYATTACK_SILENCED or ACT_VM_PRIMARYATTACK
   372→        elseif baseAct == ACT_VM_SECONDARYATTACK then
   373→            fallbackActivity = ACT_VM_SECONDARYATTACK_SILENCED or ACT_VM_SECONDARYATTACK
   374→        end
   375→    end
   376→
   377→    self:EasySendWeaponAnim(lookupKey, fallbackActivity)
   378→
   379→    local vm = IsValid(self.Owner) and self.Owner:GetViewModel() or nil
   380→    if IsValid(vm) then
   381→        vm:SetCycle(0)
   382→        vm:SetPlaybackRate(1)
   383→    end
   384→
   385→    self.Owner:SetAnimation(PLAYER_ATTACK1)
   386→    self.Owner:MuzzleFlash()
   387→
   388→    local fireSound = self:GetShootSound()
   389→    self:EmitSound(fireSound, 110, 100, 1, CHAN_WEAPON)
   390→
   391→    self:SetNextPrimaryFire(CurTime() + self.Primary.Delay)
   392→    self:SetNextSecondaryFire(CurTime() + self.Primary.Delay)
   393→    self:TakePrimaryAmmo(self.Primary.TakeAmmo)
   394→    self.NextReload = CurTime() + 0.5
   395→    self:PostShoot()
   396→end

   398→-- ============================================================
   399→-- ShootBullets — FIXED: local spread (was global)
   400→-- ============================================================
   401→function SWEP:ShootBullets(pos, buldir, dmg, tries)
   402→    if tries <= 0 then return end
   403→    local owner = self.Owner
   404→
   405→    -- FIXED: local spread (was undeclared global)
   406→    local spread
   407→    local movement = Vector(owner:GetVelocity().x, owner:GetVelocity().y, 0):LengthSqr()
   408→    local movepercent = math.Clamp(movement / owner:GetRunSpeed()^2, 0, 1)
   409→    local move = movepercent * 0.1
   410→
   411→    if owner:OnGround() then
   412→        if self:GetUHBool("Zooming") and owner:Crouching() then
   413→            spread = self.Primary.Spread * 0.18
   414→        elseif self:GetUHBool("Zooming") then
   415→            spread = self.Primary.Spread * 0.26
   416→        elseif owner:Crouching() then
   417→            spread = self.Primary.Spread * 0.34
   418→        else
   419→            spread = self.Primary.Spread * 0.5
   420→        end
   421→    else
   422→        spread = self.Primary.Spread * 0.62
   423→    end
   424→    spread = spread + move
   425→
   426→    local bullet = {}
   427→    bullet.Num = self.Primary.NumberofShots
   428→    bullet.Src = pos
   429→    bullet.Dir = buldir
   430→    bullet.Spread = Vector(spread, spread, 0)
   431→    bullet.Tracer = 1
   432→    bullet.TracerName = "uh_tracer"
   433→    bullet.Force = self.Primary.Force
   434→    bullet.Damage = dmg
   435→    bullet.AmmoType = self.Primary.Ammo
   436→    bullet.Callback = function(attacker, bultrace, dmginfo)
   437→        local mat = bultrace.MatType
   438→        if SERVER then
   439→            util.ScreenShake(bultrace.HitPos, 5, 0.1, 0.5, 64)
   440→        end
   441→        if bultrace.Hit then
   442→            local dir = bultrace.HitNormal
   443→            local tr = {}
   444→            tr.start = bultrace.HitPos - dir * (self.PenetrationDepth or 4)
   445→            tr.endpos = bultrace.HitPos
   446→            tr.filter = owner
   447→            tr.mask = MASK_SHOT
   448→            local trace = util.TraceLine(tr)
   449→            if not trace.AllSolid and trace.Fraction > 0 and GetConVar("uh_sv_penetration"):GetBool() then
   450→                self:ShootBullets(trace.HitPos, buldir, math.Round(dmg * math.Rand(0.5, 0.6)), tries - 1)
   451→            end
   452→            if mat == MAT_METAL or mat == MAT_VENT or mat == MAT_GRATE then
   453→                local fx = EffectData()
   454→                fx:SetOrigin(bultrace.HitPos)
   455→                fx:SetScale(1)
   456→                util.Effect("uh_hitworld", fx)
   457→            end
   458→        end
   459→    end
   460→
   461→    owner:FireBullets(bullet)
   462→end

   488→-- CanPrimaryAttack — FIXED: UH_GrenadeTime (was GrenadeTime)
   489→-- ============================================================
   490→function SWEP:CanPrimaryAttack()
   491→    if not IsValid(self.Owner) then return false end
   492→    if self:GetNWInt("FireMode") == 0
   493→       or self:GetNWFloat("DeployTime") > CurTime()
   494→       or self:GetUHBool("Running")
   495→       or self:GetUHBool("Reloading") then return false end
   496→    if self:Clip1() <= 0 then
   497→        if not self:GetUHBool("Reloading") then
   498→            self:EmitSound("Weapon_SMG1.Empty", 75, 100, 1, CHAN_USER_BASE)
   499→        end
   500→        self:SetNextPrimaryFire(CurTime() + 0.4)
   501→        return false
   502→    end
   503→    return true
   504→end

   514→function SWEP:SecondaryAttack()
   515→    local ct = CurTime()
   516→    local sp = game.SinglePlayer()
   517→
   518→    if self:GetUHBool("Reloading") or self:GetNWFloat("DeployTime") > ct then return end
   519→
   520→    if SERVER or iftp2() then
   521→        if self.Owner:KeyDown(IN_USE) then
   522→            local mode = self:GetNWInt("FireMode") + 1
   523→            if mode > #self.FireModes then mode = 0 end
   524→
   525→            local old = self.FireModes[self:GetNWInt("FireMode")]
   526→            if old and old.holster then
   527→                old.holster(self.Owner, self)
   528→                if sp and SERVER then
   529→                    net.Start("UH_Select_Fire")
   530→                        net.WriteFloat(self:GetNWInt("FireMode"))
   531→                        net.WriteBool(false)
   532→                    net.Broadcast()
   533→                end
   534→            elseif self:GetNWInt("FireMode") == 0 then
   535→                self:SendWeaponAnim(ACT_VM_LOWERED_TO_IDLE)
   536→            end
   537→
   538→            local new = self.FireModes[mode]
   539→            if new and new.equip then
   540→                new.equip(self.Owner, self)
   541→                if sp and SERVER then
   542→                    net.Start("UH_Select_Fire")
   543→                        net.WriteFloat(mode)
   544→                        net.WriteBool(true)
   545→                    net.Broadcast()
   546→                end
   547→            elseif mode == 0 then
   548→                self:SetUHBool("Running", false)
   549→                self:SetUHBool("Zooming", false)
   550→                self:SendWeaponAnim(ACT_VM_IDLE_TO_LOWERED)
   551→            end
   552→
   553→            self:SetNextPrimaryFire(ct + 0.5)
   554→            self:SetNextSecondaryFire(ct + 0.2)
   555→            self:SetNWInt("FireMode", mode)

   702→function SWEP:Reload()
   703→    local ct = CurTime()
   704→    if not IsValid(self.Owner) then return end
   705→
   706→    -- FIXED: UH_GrenadeTime (was GrenadeTime)
   707→    if self.NextReload < ct and not self:GetUHBool("Running") and not self:GetUHBool("Reloading")
   708→       and not self.Owner:KeyDown(IN_USE)
   709→       and self.Owner:GetNWFloat("UH_GrenadeTime") < ct
   710→       and self:GetNWFloat("DeployTime") < ct then
   711→
   712→        if self:Clip1() < (self.Chambering and self.Primary.ClipSize + 1 or self.Primary.ClipSize)
   713→           and self.Owner:GetAmmoCount(self.Primary.Ammo) > 0 then
   714→
   715→            if SERVER then
   716→                self.Owner:SetAnimation(PLAYER_RELOAD)
   717→            end
   718→            if self:GetNWInt("FireMode") == 0 then self:SetNWInt("FireMode", 1) end
   719→
   720→            self:PreReload()
   721→
   722→            -- Play reload animation
   723→            local vm = IsValid(self.Owner) and self.Owner:GetViewModel() or nil
   724→            local reloadAnimKey = nil
   725→            local pickedVariant = nil
   726→
   727→            if self.ReloadAnim and type(self.ReloadAnim) == "string" then
   728→                local seq = IsValid(vm) and vm:LookupSequence(self.ReloadAnim) or -1
   729→                if seq and seq >= 0 then
   730→                    self:SendSequence(vm, seq)
   731→                end
   732→            else
   733→                local isSilenced = self:GetNWBool("Silenced")
   734→                local isEmpty = (self:Clip1() <= 0)
   735→                local animToPlay = nil
   736→
   737→                local function GetBestAnim(emptyKey, normalKey)
   738→                    local result
   739→                    if isSilenced then
   740→                        result = self.Animations[emptyKey.."_sil"] or self.Animations[emptyKey]
   741→                            or self.Animations[normalKey.."_sil"] or self.Animations[normalKey]
   742→                    else
   743→                        result = self.Animations[emptyKey] or self.Animations[normalKey]
   744→                    end
   745→                    if istable(result) then
   746→                        pickedVariant = result[math.random(1, #result)]
   747→                        result = pickedVariant
   748→                    end
   749→                    return result
   750→                end
   751→
   752→                if isEmpty then
   753→                    animToPlay = GetBestAnim("reload_empty", "reload")
   754→                    reloadAnimKey = "reload_empty"
   755→                else
   756→                    animToPlay = GetBestAnim("reload", "reload")
   757→                    reloadAnimKey = "reload"
   758→                end
   759→
   760→                if animToPlay then
   761→                    self:SendAnim(vm, animToPlay, ACT_VM_RELOAD)
   762→                else
   763→                    if self:GetNWBool("Silenced") then
   764→                        self:SendSequence(vm, vm:SelectWeightedSequence(ACT_VM_RELOAD_SILENCED or ACT_VM_RELOAD))
   765→                    else
   766→                        self:SendSequence(vm, vm:SelectWeightedSequence(ACT_VM_RELOAD))
   767→                    end
   768→                end
   769→            end
   770→
   771→            -- Reload duration
   772→            local AnimTime = self.ReloadTime or (IsValid(vm) and vm:SequenceDuration() or 2)
   773→            local speed = self.ReloadSpeed or 1
   774→            if speed ~= 1 then
   775→                AnimTime = AnimTime / speed
   776→                if IsValid(vm) then vm:SetPlaybackRate(speed) end
   777→            else
   778→                if IsValid(vm) then vm:SetPlaybackRate(1) end
   779→            end
   780→
   781→            self._reloadFinished = false
   782→            self._reloadEndTime = ct + AnimTime
   783→
   784→            -- Setup AnimSounds
   785→            if reloadAnimKey then
   786→                self:SetupAnimSounds(reloadAnimKey, pickedVariant)
   787→            end
   788→
   789→            self.NextReload = ct + AnimTime + 0.5
   790→            self:SetNextPrimaryFire(ct + AnimTime)
   791→            self:SetUHBool("Reloading", true)
   792→            self:SetUHBool("Zooming", false)
   793→            self:SetNWFloat("ReloadTime", AnimTime)
   794→            self:SetNWFloat("ReloadEndTime", ct + AnimTime)
   795→
   796→            -- ReloadTable sounds (if no AnimSounds entry exists)
   797→            if self.UseReloadTable and not self._animSoundActive then
   798→                local sp2 = game.SinglePlayer()
   799→                local iftp = IsFirstTimePredicted()
   800→                if (sp2 and SERVER) or (not sp2 and CLIENT and iftp) then
   801→                    local scale = 1 / speed
   802→                    for _, tbl in ipairs(self.ReloadTable) do
   803→                        if tbl and tbl.delay then
   804→                            -- FIXED: was `return` (exited entire Reload function)
   805→                            -- Now uses proper continue pattern
   806→                            timer.Simple(tbl.delay * scale, function()
   807→                                if not IsValid(self) or not self:GetUHBool("Reloading") then return end
   808→                                local owner = self:GetOwner()
   809→                                if not IsValid(owner) or owner:GetActiveWeapon() != self then return end
   810→                                if tbl.sound and tbl.sound ~= "" and tbl.sound ~= "nil" then
   811→                                    local snd = tbl.sound
   812→                                    if istable(snd) then snd = snd[math.random(1, #snd)] end
   813→                                    owner:EmitSound(snd, tbl.level or 75, tbl.pitch or 100, 1, CHAN_USER_BASE)
   814→                                end
   815→                            end)
   816→                        end
   817→                    end
   818→                end
   819→            end
   820→        end
   821→    end
   822→end
