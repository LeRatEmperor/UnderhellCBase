   171→   171→   171→function SWEP:PrimaryAttack()
   172→   172→   172→    -- Block during melee or mantle animation
   173→   173→   173→    if self._meleeActive then return end
   174→   174→   174→    if self._mantleActive then return end
   175→   175→   175→
   176→   176→   176→    -- E + M1 = melee
   177→   177→   177→    if self.Owner:KeyDown(IN_USE) then
   178→   178→   178→        local ct = CurTime()
   179→   179→   179→        if ct < (self._nextMelee or 0) then return end
   180→   180→   180→        if self:GetUHBool("Reloading") then return end
   181→   181→   181→        if self:GetNWFloat("DeployTime") > ct then return end
   182→   182→   182→        if self:GetNWInt("FireMode") == 0 then return end
   183→   183→   183→
   184→   184→   184→        if SERVER or IsFirstTimePredicted() then
   185→   185→   185→            self:MeleeAttack()
   186→   186→   186→        end
   187→   187→   187→        return
   188→   188→   188→    end
   189→   189→   189→
   190→   190→   190→    -- Normal shooting — delegate to base
   191→   191→   191→    return BaseClass.PrimaryAttack(self)

   196→   196→   196→-- MELEE ATTACK
   197→   197→   197→-- ==========================================
   198→   198→   198→-- Plays ["melee"] animation, schedules a hit trace at MeleeHitDelay,
   199→   199→   199→-- and locks controls for the animation duration.
   200→   200→   200→
   201→   201→   201→function SWEP:MeleeAttack()
   202→   202→   202→    local ct = CurTime()
   203→   203→   203→    local sp = game.SinglePlayer()
   204→   204→   204→    local iftp = IsFirstTimePredicted()
   205→   205→   205→
   206→   206→   206→    if not (sp or iftp) then return end
   207→   207→   207→
   208→   208→   208→    -- Cancel conflicting states
   209→   209→   209→    self:SetUHBool("Zooming", false)
   210→   210→   210→
   211→   211→   211→    if self:GetUHBool("Running") then
   212→   212→   212→        self:SetUHBool("Running", false)
   213→   213→   213→    end
   214→   214→   214→
   215→   215→   215→    -- Optionally cancel reload
   216→   216→   216→    if self.MeleeInterruptReload ~= false and self:GetUHBool("Reloading") then
   217→   217→   217→        self:SetUHBool("Reloading", false)
   218→   218→   218→        self:SetNWFloat("ReloadTime", 0)
   219→   219→   219→        self:SetNWFloat("ReloadEndTime", 0)
   220→   220→   220→        if timer.Exists("UHReload_"..self.Owner:SteamID()) then
   221→   221→   221→            timer.Remove("UHReload_"..self.Owner:SteamID())
   222→   222→   222→        end
   223→   223→   223→    end
   224→   224→   224→
   225→   225→   225→    -- Set melee state
   226→   226→   226→    self._meleeActive = true
   227→   227→   227→    self._engineWantsIdle = nil
   228→   228→   228→    self._customIdleActive = false
   229→   229→   229→    self:ClearAnimSounds()
   230→   230→   230→
   231→   231→   231→    -- Play ["melee"] animation via EasySendWeaponAnim
   232→   232→   232→    -- Falls back to ACT_VM_MELEE if the key doesn't exist.
   233→   233→   233→    self:EasySendWeaponAnim("melee", ACT_VM_MELEE)
   234→   234→   234→
   235→   235→   235→    local vm = self.Owner:GetViewModel()
   236→   236→   236→    local animDuration = IsValid(vm) and vm:SequenceDuration() or 0.5
   237→   237→   237→
   238→   238→   238→    -- Timing
   239→   239→   239→    self._meleeHitTime = ct + (self.MeleeHitDelay or 0.15)
   240→   240→   240→    self._meleeHitDone = false
   241→   241→   241→    self._meleeEndTime = ct + animDuration
   242→   242→   242→    self._nextMelee = ct + (self.MeleeDelay or 0.6)
   243→   243→   243→
   244→   244→   244→    -- Lock controls
   245→   245→   245→    self:SetNextPrimaryFire(ct + animDuration)
   246→   246→   246→    self:SetNextSecondaryFire(ct + animDuration)
   247→   247→   247→    self.NextReload = ct + animDuration
   248→   248→   248→
   249→   249→   249→    -- Play swing sound
   250→   250→   250→    self:PlayMeleeSound(self.MeleeSound, 75, 100)
   251→   251→   251→end

   253→   253→   253→function SWEP:PlayMeleeSound(soundEntry, vol, pitch)
   254→   254→   254→    -- Supports both single strings and tables of strings (random pick)
   255→   255→   255→    if not soundEntry then return end
   256→   256→   256→    local snd = soundEntry
   257→   257→   257→    if istable(soundEntry) then
   258→   258→   258→        snd = soundEntry[math.random(1, #soundEntry)]
   259→   259→   259→    end
   260→   260→   260→    if snd and snd ~= "" then
   261→   261→   261→        self:EmitSound(snd, vol or 75, pitch or 100, 1, CHAN_USER_BASE)
   262→   262→   262→    end
   263→   263→   263→end

   265→   265→   265→-- ==========================================
   266→   266→   266→-- MELEE HIT TRACE
   267→   267→   267→-- ==========================================
   268→   268→   268→-- Hull trace forward. Applies damage to valid targets.
   269→   269→   269→-- Called from Think() at MeleeHitDelay seconds after melee start.
   270→   270→   270→
   271→   271→   271→function SWEP:DoMeleeTrace()
   272→   272→   272→    local ply = self.Owner
   273→   273→   273→    if not IsValid(ply) then return end
   274→   274→   274→
   275→   275→   275→    local pos = ply:GetShootPos()
   276→   276→   276→    local aim = ply:GetAimVector()
   277→   277→   277→    local range = self.MeleeRange or 64
   278→   278→   278→
   279→   279→   279→    local tr = util.TraceHull({
   280→   280→   280→        start = pos,
   281→   281→   281→        endpos = pos + aim * range,
   282→   282→   282→        filter = ply,
   283→   283→   283→        mins = Vector(-10, -10, -10),
   284→   284→   284→        maxs = Vector(10, 10, 10),
   285→   285→   285→        mask = MASK_SHOT_HULL,
   286→   286→   286→    })
   287→   287→   287→
   288→   288→   288→    if tr.Hit then
   289→   289→   289→        local target = tr.Entity
   290→   290→   290→
   291→   291→   291→        if IsValid(target) and SERVER then
   292→   292→   292→            local dmg = DamageInfo()
   293→   293→   293→            dmg:SetDamage(self.MeleeDamage or 50)
   294→   294→   294→            dmg:SetAttacker(ply)
   295→   295→   295→            dmg:SetInflictor(self)
   296→   296→   296→            dmg:SetDamageForce(aim * (self.MeleeForce or 300))
   297→   297→   297→            dmg:SetDamagePosition(tr.HitPos)
   298→   298→   298→            dmg:SetDamageType(DMG_CLUB)
   299→   299→   299→
   300→   300→   300→            target:TakeDamageInfo(dmg)
   301→   301→   301→        end
   302→   302→   302→
   303→   303→   303→        if SERVER then
   304→   304→   304→            util.ScreenShake(tr.HitPos, 3, 0.1, 0.3, 32)
   305→   305→   305→        end
   306→   306→   306→
   307→   307→   307→        self:PlayMeleeSound(self.MeleeHitSound, 75, 100)
   308→   308→   308→    else
   309→   309→   309→        self:PlayMeleeSound(self.MeleeMissSound, 65, 100)
   310→   310→   310→    end
   311→   311→   311→
   312→   312→   312→    ply:ViewPunch(self.MeleeViewPunch or Angle(-3, 0, 0))
   313→   313→   313→end

   344→   344→   344→-- RELOAD (fully self-contained — does NOT call BaseClass.Reload)
   345→   345→   345→-- ==========================================
   346→   346→   346→-- Same fix as the MOG 12: the BO3 base shotty's cycle-based
   347→   347→   347→-- ReloadShotgun doesn't work with these models. We override both
   348→   348→   348→-- Reload() and ReloadShotgun() to use a timer-based approach.
   349→   349→   349→
   350→   350→   350→function SWEP:Reload()
   351→   351→   351→    if self._mantleActive then return end
   352→   352→   352→    if self._meleeActive then return end
   353→   353→   353→
   354→   354→   354→    -- Block reload while holding E (USE) so inspect (E+R) doesn't cancel reload
   355→   355→   355→    if self.Owner:KeyDown(IN_USE) then return end
   356→   356→   356→
   357→   357→   357→    local ct = CurTime()
   358→   358→   358→
   359→   359→   359→    -- Don't reload if already reloading, sprinting, or blocked
   360→   360→   360→    if self.NextReload >= ct then return end
   361→   361→   361→    if self:GetUHBool("Reloading") then return end
   362→   362→   362→    if self:GetUHBool("Running") then return end
   363→   363→   363→    if self.Owner:GetNWFloat("GrenadeTime", 0) > ct then return end
   364→   364→   364→    if self:GetNWFloat("DeployTime", 0) > ct then return end
   365→   365→   365→
   366→   366→   366→    -- Don't reload if clip is full or no reserve ammo
   367→   367→   367→    if self:Clip1() >= self.Primary.ClipSize then return end
   368→   368→   368→    if self.Owner:GetAmmoCount(self:GetPrimaryAmmoType()) <= 0 then return end
   369→   369→   369→
   370→   370→   370→    -- Ensure fire mode is active
   371→   371→   371→    if self:GetNWInt("FireMode") == 0 then self:SetNWInt("FireMode", 1) end
   372→   372→   372→
   373→   373→   373→    -- Play third-person reload animation
   374→   374→   374→    self.Owner:SetAnimation(PLAYER_RELOAD)
   375→   375→   375→    self.NextReload = ct + 0.5
   376→   376→   376→
   377→   377→   377→    -- Call PreReload (sets b_clip for pump logic)
   378→   378→   378→    self:PreReload()
   379→   379→   379→
   380→   380→   380→    -- Handle flashlight/flare cancellation (server-side)
   381→   381→   381→    if SERVER then
   382→   382→   382→        if self.Owner:GetNWBool("UH_Flashlight") then
   383→   383→   383→            self.Owner:SetNWBool("UH_Flashlight", false)

   488→   488→   488→        end

   490→   490→   490→        -- Insert shell (server-authoritative)

   619→   619→   619→        self._holstering = nil
   620→   620→   620→        self._holsterFinish = nil
   621→   621→   621→        if SERVER or IsFirstTimePredicted() then

   625→   625→   625→                self.Owner:SelectWeapon(target:GetClass())

   634→   634→   634→        self:ReloadShotgun(ct)

   702→   702→   702→function SWEP:HandleRunning( ct )
   703→   703→   703→    if self:GetUHBool("Reloading") then
   704→   704→   704→        return
   705→   705→   705→    end
   706→   706→   706→        
   707→   707→   707→        local vm = self.Owner:GetViewModel()
   708→   708→   708→    if IsValid(vm) then
   709→   709→   709→        local fireDelay = self:GetNextPrimaryFire() - ct
   710→   710→   710→        if fireDelay > 0.3 then return end
   711→   711→   711→    end
   712→   712→   712→        
   713→   713→   713→        if self:GetNWFloat("DeployTime", 10) then
   714→   714→   714→

   718→   718→   718→    local isSafeMode = self:GetNWInt("FireMode") == 0
   719→   719→   719→
   720→   720→   720→    -- 1. SPRINTING (Takes Priority)

   722→   722→   722→        self:SetHoldType( self.PassiveAnim )
   723→   723→   723→        self:SetUHBool("Running", true) -- Set to TRUE so animations play

   728→   728→   728→            self.NextReload = ct + 0.5
   729→   729→   729→            if timer.Exists("UHReload_"..self.Owner:SteamID()) then
   730→   730→   730→                timer.Remove( "UHReload_"..self.Owner:SteamID() )
   731→   731→   731→            end
   732→   732→   732→        end
   733→   733→   733→
   734→   734→   734→    -- 2. NOT SPRINTING
   735→   735→   735→                else
   736→   736→   736→                        self:SetHoldType( self.HoldType )
   737→   737→   737→        
   738→   738→   738→                        -- If we are in Safe Mode (and not sprinting), ensure we are flagged as NOT running.
   739→   739→   739→                        -- This allows the procedural lowering to take effect.
   740→   740→   740→                        if isSafeMode then
   741→   741→   741→                                self:SetUHBool("Running", false) 
   742→   742→   742→                        else
   743→   743→   743→                                self:SetUHBool("Running", false)
   744→   744→   744→                        end
   745→   745→   745→                end
   746→   746→   746→        end

   749→   749→   749→function SWEP:GetViewModelPosition(pos, ang)
   750→   750→   750→    local ft = FrameTime()
   751→   751→   751→
   752→   752→   752→    -- Pass through BaseClass logic (Sway, Inspect, etc.)
   753→   753→   753→    if BaseClass and BaseClass.GetViewModelPosition then
   754→   754→   754→        pos, ang = BaseClass.GetViewModelPosition(self, pos, ang)
   755→   755→   755→    end
   756→   756→   756→
   757→   757→   757→    -- ==========================================
   758→   758→   758→    -- HYBRID LOWERING LOGIC (Safe Mode)
   759→   759→   759→    -- ==========================================
   760→   760→   760→    
   761→   761→   761→    -- We only use procedural LoweredPos/Ang if we are in SAFE MODE
   762→   762→   762→    -- AND we are NOT Sprinting.
   763→   763→   763→    local target = 0
   764→   764→   764→    if self:GetNWInt("FireMode") == 0 and not self:GetUHBool("Running") then
   765→   765→   765→        target = 1
   766→   766→   766→    end
   767→   767→   767→
   768→   768→   768→    self._uhLower = Lerp(ft * 8, self._uhLower or 0, target)
   769→   769→   769→
   770→   770→   770→    -- Apply Safe Mode Lowering (Procedural)
   771→   771→   771→    if self._uhLower > 0.001 then
   772→   772→   772→        local lp = self.LoweredPos or vector_origin
   773→   773→   773→        local la = self.LoweredAng or angle_zero
   774→   774→   774→
   775→   775→   775→        local ap, ay, ar = 0, 0, 0
   776→   776→   776→        if isangle(la) then
   777→   777→   777→            ap, ay, ar = la.p, la.y, la.r
   778→   778→   778→        elseif isvector(la) then
   779→   779→   779→            ap, ay, ar = la.x, la.y, la.z
   780→   780→   780→        end
   781→   781→   781→
   782→   782→   782→        ang:RotateAroundAxis(ang:Right(),   ap * self._uhLower)
   783→   783→   783→        ang:RotateAroundAxis(ang:Up(),      ay * self._uhLower)
   784→   784→   784→        ang:RotateAroundAxis(ang:Forward(), ar * self._uhLower)
   785→   785→   785→
   786→   786→   786→        pos = pos
   787→   787→   787→            + ang:Right()   * lp.x * self._uhLower
   788→   788→   788→            + ang:Forward() * lp.y * self._uhLower
   789→   789→   789→            + ang:Up()      * lp.z * self._uhLower
   790→   790→   790→    end
   791→   791→   791→
   792→   792→   792→    -- ==========================================
   793→   793→   793→    -- ALTERNATIVE POS/ANG LOGIC (Ultra Guns Style)
   794→   794→   794→    -- ==========================================
   795→   795→   795→    
   796→   796→   796→    -- Determine if we should show the offset
   797→   797→   797→    -- We hide it if: Sprinting, Zooming, or in Safe Mode
   798→   798→   798→    local targetAlt = 1
   799→   799→   799→    if self:GetUHBool("Running") or self:GetUHBool("Zooming") or self:GetNWInt("FireMode") == 0 then
   800→   800→   800→        targetAlt = 0
   801→   801→   801→    end
   802→   802→   802→
   803→   803→   803→    -- Smoothly interpolate the factor
   804→   804→   804→    self._altFactor = Lerp(ft * 10, self._altFactor or 0, targetAlt)
   805→   805→   805→
   806→   806→   806→    -- Apply the offset if it exists
   807→   807→   807→    if (self.AlternativePos or self.AlternativeAng) and self._altFactor > 0.01 then
   808→   808→   808→        local ap = self.AlternativePos or vector_origin
   809→   809→   809→        local aa = self.AlternativeAng or angle_zero
   810→   810→   810→
   811→   811→   811→        -- Apply Position Offset
   812→   812→   812→        pos = pos
   813→   813→   813→            + ang:Right()   * ap.x * self._altFactor
   814→   814→   814→            + ang:Forward() * ap.y * self._altFactor
   815→   815→   815→            + ang:Up()      * ap.z * self._altFactor
   816→   816→   816→
   817→   817→   817→        -- Apply Angle Offset
   818→   818→   818→        local ap_p, ap_y, ap_r = 0, 0, 0
   819→   819→   819→        if isangle(aa) then
   820→   820→   820→            ap_p, ap_y, ap_r = aa.p, aa.y, aa.r
   821→   821→   821→        elseif isvector(aa) then
   822→   822→   822→            ap_p, ap_y, ap_r = aa.x, aa.y, aa.z
   823→   823→   823→        end
   824→   824→   824→
   825→   825→   825→        ang:RotateAroundAxis(ang:Right(),   ap_p * self._altFactor)
   826→   826→   826→        ang:RotateAroundAxis(ang:Up(),      ap_y * self._altFactor)
   827→   827→   827→        ang:RotateAroundAxis(ang:Forward(), ap_r * self._altFactor)
   828→   828→   828→    end
   829→   829→   829→
   830→   830→   830→    -- ==========================================
   831→   831→   831→    -- SPRINT Z-AXIS OFFSET (Smoothed)
   832→   832→   832→    -- ==========================================
   833→   833→   833→    local targetZ = self:GetUHBool("Running") and 0 or 0
   834→   834→   834→    self._sprintZOffset = Lerp(ft * 10, self._sprintZOffset or 0, targetZ)
   835→   835→   835→    pos = pos + ang:Up() * self._sprintZOffset
   836→   836→   836→
   837→   837→   837→    return pos, ang
   838→   838→   838→end

   840→   840→   840→function SWEP:HandleSprintingAnimations()
   841→   841→   841→    local ply = self:GetOwner()
   842→   842→   842→    
   843→   843→   843→    if not IsValid(ply) or not ply:IsPlayer() then return end
   844→   844→   844→
   845→   845→   845→    -- Use UH variable for running state
   846→   846→   846→    local isRunning = self:GetUHBool("Running")
   847→   847→   847→    local isReloading = self:GetUHBool("Reloading")
   848→   848→   848→
   849→   849→   849→    -- Initialize tracking variable
   850→   850→   850→    if self.wasRunning == nil then self.wasRunning = false end
   851→   851→   851→
   852→   852→   852→    local vm = ply:GetViewModel()
   853→   853→   853→    if not IsValid(vm) then return end
   854→   854→   854→
   855→   855→   855→    -- TRANSITION: Enter Sprint
   856→   856→   856→    if isRunning and not self.wasRunning then
   857→   857→   857→        if not isReloading then
   858→   858→   858→            self:EasySendWeaponAnim("sprint_in", ACT_VM_SPRINT_ENTER)
   859→   859→   859→        end
   860→   860→   860→    
   861→   861→   861→    -- TRANSITION: Exit Sprint
   862→   862→   862→    elseif not isRunning and self.wasRunning then
   863→   863→   863→        if not isReloading then
   864→   864→   864→            self:EasySendWeaponAnim("sprint_out", ACT_VM_SPRINT_LEAVE)
   865→   865→   865→        end
   866→   866→   866→    
   867→   867→   867→    -- LOOPING: While Sprinting (and not reloading)
   868→   868→   868→    elseif isRunning and not isReloading then
   869→   869→   869→        -- If the current sprint animation finished (cycle >= 1), restart it
   870→   870→   870→        if vm:GetCycle() >= 1 then
   871→   871→   871→            self:EasySendWeaponAnim("sprint_idle", ACT_VM_SPRINT_IDLE)
   872→   872→   872→        end
   873→   873→   873→    end
   874→   874→   874→
   875→   875→   875→    self.wasRunning = isRunning
   876→   876→   876→end
