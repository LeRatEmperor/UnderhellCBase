   146→function SWEP:Holster(wep)
   147→    -- Cancel any in-flight burst
   148→    self._burstRemaining = nil
   149→    self._burstNextFire = nil
   150→    -- Cancel rechamber
   151→    self._isRechambering = false
   152→    -- Reset animation transition trackers
   153→    self.wasZooming = false
   154→    self.wasRunning = false
   155→    -- Cancel melee
   156→    if self._meleeActive then
   157→        self._meleeActive = false
   158→        self._meleeHitTime = nil
   159→        self._meleeHitDone = nil
   160→        self._meleeEndTime = nil
   161→        self:ClearAnimSounds()
   162→    end
   163→    -- Cancel mantle
   164→    if self._mantleActive then
   165→        self._mantleActive = false
   166→        self._mantleEndTime = nil
   167→        self:ClearAnimSounds()
   168→    end
   169→    -- Note: timer.Simple closures will still fire but the above
   170→    -- state resets make their checks self-correcting
   171→    return BaseClass.Holster(self, wep)
   172→end

   174→function SWEP:HandleRunning( ct )
   175→    if self:GetUHBool("Reloading") then
   176→        return
   177→    end
   178→        
   179→        local vm = self.Owner:GetViewModel()
   180→    if IsValid(vm) then
   181→        local fireDelay = self:GetNextPrimaryFire() - ct
   182→        if fireDelay > 0.3 then return end
   183→    end
   184→        
   185→        if self:GetNWFloat("DeployTime", 10) then
   186→
   187→
   188→    local dist = self.Owner:GetVelocity():LengthSqr()
   189→    local isSprinting = self.Owner:KeyDown( IN_SPEED ) and dist > self.Owner:GetWalkSpeed()^2
   190→    local isSafeMode = self:GetNWInt("FireMode") == 0
   191→
   192→    -- 1. SPRINTING (Takes Priority)
   193→    if isSprinting then
   194→        self:SetHoldType( self.PassiveAnim )
   195→        self:SetUHBool("Running", true) -- Set to TRUE so animations play
   196→        self:SetUHBool("Zooming", false)
   197→        
   198→        if self:GetUHBool("Reloading") then
   199→            self:SetUHBool("Reloading", false)
   200→            self.NextReload = ct + 0.5
   201→            if timer.Exists("UHReload_"..self.Owner:SteamID()) then
   202→                timer.Remove( "UHReload_"..self.Owner:SteamID() )
   203→            end
   204→        end
   205→
   206→    -- 2. NOT SPRINTING
   207→                else
   208→                        self:SetHoldType( self.HoldType )
   209→        
   210→                        -- If we are in Safe Mode (and not sprinting), ensure we are flagged as NOT running.
   211→                        -- This allows the procedural lowering to take effect.
   212→                        if isSafeMode then
   213→                                self:SetUHBool("Running", false) 
   214→                        else
   215→                                self:SetUHBool("Running", false)
   216→                        end
   217→                end
   218→        end
   219→end

   225→function SWEP:StartMantle()
   226→    local ct = CurTime()
   227→    local sp = game.SinglePlayer()
   228→    local iftp = IsFirstTimePredicted()
   229→
   230→    if not (sp or iftp) then return end
   231→
   232→    self:SetUHBool("Zooming", false)
   233→    if self:GetUHBool("Running") then
   234→        self:SetUHBool("Running", false)
   235→    end
   236→    if self._meleeActive then
   237→        self:EndMelee()
   238→    end
   239→
   240→    -- Cancel reload
   241→    if self:GetUHBool("Reloading") then
   242→        self:SetUHBool("Reloading", false)
   243→        self:SetNWFloat("ReloadTime", 0)
   244→        self:SetNWFloat("ReloadEndTime", 0)
   245→        if timer.Exists("UHReload_"..self.Owner:SteamID()) then
   246→            timer.Remove("UHReload_"..self.Owner:SteamID())
   247→        end
   248→    end
   249→
   250→    -- Cancel burst
   251→    self._burstRemaining = nil
   252→    self._burstNextFire = nil
   253→
   254→    self._mantleActive = true
   255→    self._engineWantsIdle = nil
   256→    self._customIdleActive = false
   257→    self:ClearAnimSounds()
   258→
   259→    self:EasySendWeaponAnim("mantle", ACT_VM_MELEE_SHOVE)
   260→
   261→    local vm = self.Owner:GetViewModel()
   262→    local animDuration = IsValid(vm) and vm:SequenceDuration() or (self.MantleDuration or 0.6)
   263→    local mantleDur = math.max(animDuration, self.MantleDuration or 0.6)
   264→    self._mantleEndTime = ct + mantleDur
   265→
   266→    self:SetNextPrimaryFire(ct + mantleDur)
   267→    self:SetNextSecondaryFire(ct + mantleDur)
   268→    self.NextReload = ct + mantleDur
   269→end

   284→function SWEP:Think()
   285→    local ct = CurTime()
   286→
   287→    -- ==========================================
   288→    -- PARKOUR TRAVERSAL DETECTION
   289→    -- ==========================================
   290→    local ply = self.Owner
   291→    if IsValid(ply) then
   292→        local isVaulting = ply:GetNW2Bool("BO3_IsVaulting", false)
   293→        local isMantling = ply:GetNW2Bool("BO3_IsMantling", false)
   294→        local isInTraversal = isVaulting or isMantling
   295→
   296→        if isInTraversal and not self._mantleActive then
   297→            if SERVER or IsFirstTimePredicted() then
   298→                self:StartMantle()
   299→            end
   300→        end
   301→
   302→        if not isInTraversal and self._mantleActive then
   303→            self:EndMantle()
   304→        end
   305→
   306→        self._lastBO3IsVaulting = isVaulting
   307→        self._lastBO3IsMantling = isMantling
   308→    end
   309→
   310→    -- Mantle timeout safety
   311→    if self._mantleActive and self._mantleEndTime and ct >= self._mantleEndTime then
   312→        self:EndMantle()
   313→    end
   314→
   315→    -- Continue burst rounds (works even if player released M1)
   316→    if self._burstRemaining and self._burstRemaining > 0 then
   317→        if ct >= (self._burstNextFire or 0) then
   318→            self:FireBurstRound()
   319→        end
   320→    end
   321→
   322→    -- Melee hit timing
   323→    if self._meleeActive and not self._meleeHitDone and self._meleeHitTime and ct >= self._meleeHitTime then
   324→        self._meleeHitDone = true
   325→        if SERVER or IsFirstTimePredicted() then
   326→            self:DoMeleeTrace()
   327→        end
   328→    end
   329→
   330→    -- Melee end timing
   331→    if self._meleeActive and self._meleeEndTime and ct >= self._meleeEndTime then
   332→        self:EndMelee()
   333→    end
   334→
   335→    BaseClass.Think(self)
   336→    self:HandleSprintingAnimations()
   337→end

   339→function SWEP:GetViewModelPosition(pos, ang)
   340→    local ft = FrameTime()
   341→
   342→    -- Pass through BaseClass logic (Sway, Inspect, etc.)
   343→    if BaseClass and BaseClass.GetViewModelPosition then
   344→        pos, ang = BaseClass.GetViewModelPosition(self, pos, ang)
   345→    end
   346→
   347→    -- ==========================================
   348→    -- HYBRID LOWERING LOGIC (Safe Mode)
   349→    -- ==========================================
   350→    
   351→    -- We only use procedural LoweredPos/Ang if we are in SAFE MODE
   352→    -- AND we are NOT Sprinting.
   353→    local target = 0
   354→    if self:GetNWInt("FireMode") == 0 and not self:GetUHBool("Running") then
   355→        target = 1
   356→    end
   357→
   358→    self._uhLower = Lerp(ft * 8, self._uhLower or 0, target)
   359→
   360→    -- Apply Safe Mode Lowering (Procedural)
   361→    if self._uhLower > 0.001 then
   362→        local lp = self.LoweredPos or vector_origin
   363→        local la = self.LoweredAng or angle_zero
   364→
   365→        local ap, ay, ar = 0, 0, 0
   366→        if isangle(la) then
   367→            ap, ay, ar = la.p, la.y, la.r
   368→        elseif isvector(la) then
   369→            ap, ay, ar = la.x, la.y, la.z
   370→        end
   371→
   372→        ang:RotateAroundAxis(ang:Right(),   ap * self._uhLower)
   373→        ang:RotateAroundAxis(ang:Up(),      ay * self._uhLower)
   374→        ang:RotateAroundAxis(ang:Forward(), ar * self._uhLower)
   375→
   376→        pos = pos
   377→            + ang:Right()   * lp.x * self._uhLower
   378→            + ang:Forward() * lp.y * self._uhLower
   379→            + ang:Up()      * lp.z * self._uhLower
   380→    end
   381→
   382→    -- ==========================================
   383→    -- ALTERNATIVE POS/ANG LOGIC (Ultra Guns Style)
   384→    -- ==========================================
   385→    
   386→    -- Determine if we should show the offset
   387→    -- We hide it if: Sprinting, Zooming, or in Safe Mode
   388→    local targetAlt = 1
   389→    if self:GetUHBool("Running") or self:GetUHBool("Zooming") or self:GetNWInt("FireMode") == 0 then
   390→        targetAlt = 0
   391→    end
   392→
   393→    -- Smoothly interpolate the factor
   394→    self._altFactor = Lerp(ft * 10, self._altFactor or 0, targetAlt)
   395→
   396→    -- Apply the offset if it exists
   397→    if (self.AlternativePos or self.AlternativeAng) and self._altFactor > 0.01 then
   398→        local ap = self.AlternativePos or vector_origin
   399→        local aa = self.AlternativeAng or angle_zero
   400→
   401→        -- Apply Position Offset
   402→        pos = pos
   403→            + ang:Right()   * ap.x * self._altFactor
   404→            + ang:Forward() * ap.y * self._altFactor
   405→            + ang:Up()      * ap.z * self._altFactor
   406→
   407→        -- Apply Angle Offset
   408→        local ap_p, ap_y, ap_r = 0, 0, 0
   409→        if isangle(aa) then
   410→            ap_p, ap_y, ap_r = aa.p, aa.y, aa.r
   411→        elseif isvector(aa) then
   412→            ap_p, ap_y, ap_r = aa.x, aa.y, aa.z
   413→        end
   414→
   415→        ang:RotateAroundAxis(ang:Right(),   ap_p * self._altFactor)
   416→        ang:RotateAroundAxis(ang:Up(),      ap_y * self._altFactor)
   417→        ang:RotateAroundAxis(ang:Forward(), ap_r * self._altFactor)
   418→    end
   419→
   420→    -- ==========================================
   421→    -- SPRINT Z-AXIS OFFSET (Smoothed)
   422→    -- ==========================================
   423→    local targetZ = self:GetUHBool("Running") and 0 or 0
   424→    self._sprintZOffset = Lerp(ft * 10, self._sprintZOffset or 0, targetZ)
   425→    pos = pos + ang:Up() * self._sprintZOffset
   426→
   427→    return pos, ang
   428→end

   430→function SWEP:HandleSprintingAnimations()
   431→    local ply = self:GetOwner()
   432→    
   433→    if not IsValid(ply) or not ply:IsPlayer() then return end
   434→
   435→    -- Use UH variable for running state
   436→    local isRunning = self:GetUHBool("Running")
   437→    local isReloading = self:GetUHBool("Reloading")
   438→
   439→    -- Initialize tracking variable
   440→    if self.wasRunning == nil then self.wasRunning = false end
   441→
   442→    local vm = ply:GetViewModel()
   443→    if not IsValid(vm) then return end
   444→
   445→    -- TRANSITION: Enter Sprint
   446→    if isRunning and not self.wasRunning then
   447→        if not isReloading then
   448→            self:EasySendWeaponAnim("sprint_in", ACT_VM_SPRINT_ENTER)
   449→        end
   450→    
   451→    -- TRANSITION: Exit Sprint
   452→    elseif not isRunning and self.wasRunning then
   453→        if not isReloading then
   454→            self:EasySendWeaponAnim("sprint_out", ACT_VM_SPRINT_LEAVE)
   455→        end
   456→    
   457→    -- LOOPING: While Sprinting (and not reloading)
   458→    elseif isRunning and not isReloading then
   459→        -- If the current sprint animation finished (cycle >= 1), restart it
   460→        if vm:GetCycle() >= 1 then
   461→            self:EasySendWeaponAnim("sprint_idle", ACT_VM_SPRINT_IDLE)
   462→        end
   463→    end
   464→
   465→    self.wasRunning = isRunning
   466→end

   468→function SWEP:FireBurstRound()
   469→    if not IsValid(self) or not IsValid(self.Owner) then return end
   470→    if not self:CanPrimaryAttack() then
   471→        self._burstRemaining = nil
   472→        return
   473→    end
   474→
   475→    local ct = CurTime()
   476→    local sp = game.SinglePlayer()
   477→    local iftp = IsFirstTimePredicted()
   478→
   479→    -- Bullets
   480→    if SERVER or iftp then
   481→        if self:GetNWBool("Silenced") then
   482→            self:ShootBullets(self.Owner:GetShootPos(), self.Owner:GetAimVector(), math.Round(math.random(self.Primary.MinDamage, self.Primary.MaxDamage) * 0.95), self.Penetration or 2)
   483→        else
   484→            self:ShootBullets(self.Owner:GetShootPos(), self.Owner:GetAimVector(), math.random(self.Primary.MinDamage, self.Primary.MaxDamage), self.Penetration or 2)
   485→        end
   486→    end
   487→
   488→    -- Recoil
   489→    local recoil = util.SharedRandom("uh_recoil", self.Primary.MinRecoil, self.Primary.MaxRecoil) * (self:GetUHBool("Zooming") and 0.35 or 1)
   490→
   491→    if sp or (CLIENT and iftp) then
   492→            self:DoMuzzleFlash()
   493→                
   494→            self:CreateSmoke( self:GetMuzzle(), self.Primary.Delay + (self.Primary.Automatic and 0.14 or 0.32) )
   495→                
   496→            if not self.NoShell then
   497→                self:CreateShell( self.ShellDelay or 0, self.ShellHeat )
   498→            end
   499→                
   500→            self.Owner:SetEyeAngles( self.Owner:EyeAngles() + Angle( recoil, 0, 0 ) )
   501→    end
   502→
   503→    self.Owner:ViewPunch(Angle(recoil, 0, 0))
   504→
   505→    -- Animation: uses ShootAnimation() which checks ["iron_fire"] (zoomed)
   506→    -- or ["fire"] (hip), falls back to ACT_VM_PRIMARYATTACK.
   507→    local shootAnim = self:ShootAnimation()
   508→    if type(shootAnim) == "string" then
   509→        self:EasySendWeaponAnim(shootAnim, ACT_VM_PRIMARYATTACK)
   510→    else
   511→        self:SendWeaponAnim(ACT_VM_PRIMARYATTACK)
   512→    end
   513→
   514→    self.Owner:SetAnimation(PLAYER_ATTACK1)
   515→    self.Owner:MuzzleFlash()
   516→
   517→    local fireSound = self:GetShootSound()
   518→    self:EmitSound(fireSound, 110, 100, 1, CHAN_WEAPON)
   519→    self:TakePrimaryAmmo(self.Primary.TakeAmmo)
   520→
   521→    self.NextReload = CurTime() + 0.5
   522→    self:PostShoot()
   523→
   524→    -- Burst timing
   525→    if SERVER or iftp then
   526→        self._burstRemaining = self._burstRemaining - 1
   527→
   528→        if self._burstRemaining > 0 then
   529→            self._burstNextFire = ct + self._burstDelay
   530→            self:SetNextPrimaryFire(ct + self._burstDelay)
   531→        else
   532→            self._burstRemaining = nil
   533→            self:SetNextPrimaryFire(ct + self.Primary.Delay * 3)
   534→            self:SetNextSecondaryFire(ct + self.Primary.Delay * 3)
   535→        end
   536→    end
   537→end

   540→-- PRIMARY ATTACK OVERRIDE
   541→-- ==========================================
   542→-- E + M1 = melee attack (Ultra Guns style)
   543→-- M1 only = normal shooting (delegates to base)
   544→
   545→function SWEP:PrimaryAttack()
   546→    -- Block during melee or mantle animation
   547→    if self._meleeActive then return end
   548→    if self._mantleActive then return end
   549→
   550→    -- E + M1 = melee
   551→    if self.Owner:KeyDown(IN_USE) then
   552→        local ct = CurTime()
   553→        if ct < (self._nextMelee or 0) then return end
   554→        if self:GetUHBool("Reloading") then return end
   555→        if self:GetNWFloat("DeployTime") > ct then return end
   556→        if self:GetNWInt("FireMode") == 0 then return end
   557→
   558→        if SERVER or IsFirstTimePredicted() then
   559→            self:MeleeAttack()
   560→        end
   561→        return
   562→    end
   563→
   564→    -- Normal shooting — delegate to base
   565→    return BaseClass.PrimaryAttack(self)
   566→end

   569→-- ==========================================
   570→-- MELEE ATTACK
   571→-- ==========================================
   572→
   573→function SWEP:MeleeAttack()
   574→    local ct = CurTime()
   575→    local sp = game.SinglePlayer()
   576→    local iftp = IsFirstTimePredicted()
   577→
   578→    if not (sp or iftp) then return end
   579→
   580→    self:SetUHBool("Zooming", false)
   581→
   582→    if self:GetUHBool("Running") then
   583→        self:SetUHBool("Running", false)
   584→    end
   585→
   586→    if self.MeleeInterruptReload ~= false and self:GetUHBool("Reloading") then
   587→        self:SetUHBool("Reloading", false)
   588→        self:SetNWFloat("ReloadTime", 0)
   589→        self:SetNWFloat("ReloadEndTime", 0)
   590→        if timer.Exists("UHReload_"..self.Owner:SteamID()) then
   591→            timer.Remove("UHReload_"..self.Owner:SteamID())
   592→        end
   593→    end
   594→
   595→    self._meleeActive = true
   596→    self._engineWantsIdle = nil
   597→    self._customIdleActive = false
   598→    self:ClearAnimSounds()
   599→
   600→    self:EasySendWeaponAnim("melee", ACT_VM_MELEE)
   601→
   602→    local vm = self.Owner:GetViewModel()
   603→    local animDuration = IsValid(vm) and vm:SequenceDuration() or 0.5
   604→
   605→    self._meleeHitTime = ct + (self.MeleeHitDelay or 0.15)
   606→    self._meleeHitDone = false
   607→    self._meleeEndTime = ct + animDuration
   608→    self._nextMelee = ct + (self.MeleeDelay or 0.6)
   609→
   610→    self:SetNextPrimaryFire(ct + animDuration)
   611→    self:SetNextSecondaryFire(ct + animDuration)
   612→    self.NextReload = ct + animDuration
   613→
   614→    self:PlayMeleeSound(self.MeleeSound, 75, 100)
   615→end

   618→-- ==========================================
   619→-- MELEE HIT TRACE
   620→-- ==========================================
   621→
   622→function SWEP:PlayMeleeSound(soundEntry, vol, pitch)
   623→    if not soundEntry then return end
   624→    local snd = soundEntry
   625→    if istable(soundEntry) then
   626→        snd = soundEntry[math.random(1, #soundEntry)]
   627→    end
   628→    if snd and snd ~= "" then
   629→        self:EmitSound(snd, vol or 75, pitch or 100, 1, CHAN_USER_BASE)
   630→    end
   631→end
   632→
   633→function SWEP:DoMeleeTrace()
   634→    local ply = self.Owner
   635→    if not IsValid(ply) then return end
   636→
   637→    local pos = ply:GetShootPos()
   638→    local aim = ply:GetAimVector()
   639→    local range = self.MeleeRange or 64
   640→
   641→    local tr = util.TraceHull({
   642→        start = pos,
   643→        endpos = pos + aim * range,
   644→        filter = ply,
   645→        mins = Vector(-10, -10, -10),
   646→        maxs = Vector(10, 10, 10),
   647→        mask = MASK_SHOT_HULL,
   648→    })
   649→
   650→    if tr.Hit then
   651→        local target = tr.Entity
   652→
   653→        if IsValid(target) and SERVER then
   654→            local dmg = DamageInfo()
   655→            dmg:SetDamage(self.MeleeDamage or 50)
   656→            dmg:SetAttacker(ply)
   657→            dmg:SetInflictor(self)
   658→            dmg:SetDamageForce(aim * (self.MeleeForce or 300))
   659→            dmg:SetDamagePosition(tr.HitPos)
   660→            dmg:SetDamageType(DMG_CLUB)
   661→
   662→            target:TakeDamageInfo(dmg)
   663→        end
   664→
   665→        if SERVER then
   666→            util.ScreenShake(tr.HitPos, 3, 0.1, 0.3, 32)
   667→        end
   668→
   669→        self:PlayMeleeSound(self.MeleeHitSound, 75, 100)
   670→    else
   671→        self:PlayMeleeSound(self.MeleeMissSound, 65, 100)
   672→    end
   673→
   674→    ply:ViewPunch(self.MeleeViewPunch or Angle(-3, 0, 0))
   675→end

   746→    self._mantleEndTime = nil
