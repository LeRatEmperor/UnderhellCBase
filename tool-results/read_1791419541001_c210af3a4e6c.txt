   138→function SWEP:PostShoot()
   139→    local ct = CurTime()
   140→    local pumpDelay = self.PumpDelay or 0.4
   141→
   142→    -- Ensure fire+sprint block lasts AT LEAST pumpDelay long
   143→    -- but don't shorten a longer Primary.Delay
   144→    self:SetNextPrimaryFire(math.max(self:GetNextPrimaryFire(), ct + pumpDelay))
   145→    self:SetNextSecondaryFire(math.max(self:GetNextSecondaryFire(), ct + pumpDelay))
   146→
   147→    timer.Simple(pumpDelay, function()
   148→        if !IsValid(self) or !IsValid(self.Owner) or !IsValid(self.Owner:GetActiveWeapon()) or self.Owner:GetActiveWeapon() != self then return end
   149→
   150→        local animKey = self:GetUHBool("Zooming") and "rechamber_ads" or "rechamber"
   151→        self:EasySendWeaponAnim(animKey, ACT_SHOTGUN_PUMP)
   152→    end)
   153→end

   171→function SWEP:PrimaryAttack()
   172→    -- Block during melee or mantle animation
   173→    if self._meleeActive then return end
   174→    if self._mantleActive then return end
   175→
   176→    -- E + M1 = melee
   177→    if self.Owner:KeyDown(IN_USE) then
   178→        local ct = CurTime()
   179→        if ct < (self._nextMelee or 0) then return end
   180→        if self:GetUHBool("Reloading") then return end
   181→        if self:GetNWFloat("DeployTime") > ct then return end
   182→        if self:GetNWInt("FireMode") == 0 then return end
   183→
   184→        if SERVER or IsFirstTimePredicted() then
   185→            self:MeleeAttack()
   186→        end
   187→        return
   188→    end
   189→
   190→    -- Normal shooting — delegate to base
   191→    return BaseClass.PrimaryAttack(self)
   192→end

   195→-- ==========================================
   196→-- MELEE ATTACK
   197→-- ==========================================
   198→-- Plays ["melee"] animation, schedules a hit trace at MeleeHitDelay,
   199→-- and locks controls for the animation duration.
   200→
   201→function SWEP:MeleeAttack()
   202→    local ct = CurTime()
   203→    local sp = game.SinglePlayer()
   204→    local iftp = IsFirstTimePredicted()
   205→
   206→    if not (sp or iftp) then return end
   207→
   208→    -- Cancel conflicting states
   209→    self:SetUHBool("Zooming", false)
   210→
   211→    if self:GetUHBool("Running") then
   212→        self:SetUHBool("Running", false)
   213→    end
   214→
   215→    -- Optionally cancel reload
   216→    if self.MeleeInterruptReload ~= false and self:GetUHBool("Reloading") then
   217→        self:SetUHBool("Reloading", false)
   218→        self:SetNWFloat("ReloadTime", 0)
   219→        self:SetNWFloat("ReloadEndTime", 0)
   220→        if timer.Exists("UHReload_"..self.Owner:SteamID()) then
   221→            timer.Remove("UHReload_"..self.Owner:SteamID())
   222→        end
   223→    end
   224→
   225→    -- Set melee state
   226→    self._meleeActive = true
   227→    self._engineWantsIdle = nil
   228→    self._customIdleActive = false
   229→    self:ClearAnimSounds()
   230→
   231→    -- Play ["melee"] animation via EasySendWeaponAnim
   232→    -- Falls back to ACT_VM_MELEE if the key doesn't exist.
   233→    self:EasySendWeaponAnim("melee", ACT_VM_MELEE)
   234→
   235→    local vm = self.Owner:GetViewModel()
   236→    local animDuration = IsValid(vm) and vm:SequenceDuration() or 0.5
   237→
   238→    -- Timing
   239→    self._meleeHitTime = ct + (self.MeleeHitDelay or 0.15)
   240→    self._meleeHitDone = false
   241→    self._meleeEndTime = ct + animDuration
   242→    self._nextMelee = ct + (self.MeleeDelay or 0.6)
   243→
   244→    -- Lock controls
   245→    self:SetNextPrimaryFire(ct + animDuration)
   246→    self:SetNextSecondaryFire(ct + animDuration)
   247→    self.NextReload = ct + animDuration
   248→
   249→    -- Play swing sound
   250→    self:PlayMeleeSound(self.MeleeSound, 75, 100)
   251→end

   253→function SWEP:PlayMeleeSound(soundEntry, vol, pitch)
   254→    -- Supports both single strings and tables of strings (random pick)
   255→    if not soundEntry then return end
   256→    local snd = soundEntry
   257→    if istable(soundEntry) then
   258→        snd = soundEntry[math.random(1, #soundEntry)]
   259→    end
   260→    if snd and snd ~= "" then
   261→        self:EmitSound(snd, vol or 75, pitch or 100, 1, CHAN_USER_BASE)
   262→    end
   263→end

   265→-- ==========================================
   266→-- MELEE HIT TRACE
   267→-- ==========================================
   268→-- Hull trace forward. Applies damage to valid targets.
   269→-- Called from Think() at MeleeHitDelay seconds after melee start.
   270→
   271→function SWEP:DoMeleeTrace()
   272→    local ply = self.Owner
   273→    if not IsValid(ply) then return end
   274→
   275→    local pos = ply:GetShootPos()
   276→    local aim = ply:GetAimVector()
   277→    local range = self.MeleeRange or 64
   278→
   279→    local tr = util.TraceHull({
   280→        start = pos,
   281→        endpos = pos + aim * range,
   282→        filter = ply,
   283→        mins = Vector(-10, -10, -10),
   284→        maxs = Vector(10, 10, 10),
   285→        mask = MASK_SHOT_HULL,
   286→    })
   287→
   288→    if tr.Hit then
   289→        local target = tr.Entity
   290→
   291→        if IsValid(target) and SERVER then
   292→            local dmg = DamageInfo()
   293→            dmg:SetDamage(self.MeleeDamage or 50)
   294→            dmg:SetAttacker(ply)
   295→            dmg:SetInflictor(self)
   296→            dmg:SetDamageForce(aim * (self.MeleeForce or 300))
   297→            dmg:SetDamagePosition(tr.HitPos)
   298→            dmg:SetDamageType(DMG_CLUB)
   299→
   300→            target:TakeDamageInfo(dmg)
   301→        end
   302→
   303→        if SERVER then
   304→            util.ScreenShake(tr.HitPos, 3, 0.1, 0.3, 32)
   305→        end
   306→
   307→        self:PlayMeleeSound(self.MeleeHitSound, 75, 100)
   308→    else
   309→        self:PlayMeleeSound(self.MeleeMissSound, 65, 100)
   310→    end
   311→
   312→    ply:ViewPunch(self.MeleeViewPunch or Angle(-3, 0, 0))
   313→end

   343→-- ==========================================
   344→-- RELOAD (fully self-contained — does NOT call BaseClass.Reload)
   345→-- ==========================================
   346→-- Same fix as the MOG 12: the BO3 base shotty's cycle-based
   347→-- ReloadShotgun doesn't work with these models. We override both
   348→-- Reload() and ReloadShotgun() to use a timer-based approach.
   349→
   350→function SWEP:Reload()
   351→    if self._mantleActive then return end
   352→    if self._meleeActive then return end
   353→
   354→    -- Block reload while holding E (USE) so inspect (E+R) doesn't cancel reload
   355→    if self.Owner:KeyDown(IN_USE) then return end
   356→
   357→    local ct = CurTime()
   358→
   359→    -- Don't reload if already reloading, sprinting, or blocked
   360→    if self.NextReload >= ct then return end
   361→    if self:GetUHBool("Reloading") then return end
   362→    if self:GetUHBool("Running") then return end
   363→    if self.Owner:GetNWFloat("GrenadeTime", 0) > ct then return end
   364→    if self:GetNWFloat("DeployTime", 0) > ct then return end
   365→
   366→    -- Don't reload if clip is full or no reserve ammo
   367→    if self:Clip1() >= self.Primary.ClipSize then return end
   368→    if self.Owner:GetAmmoCount(self:GetPrimaryAmmoType()) <= 0 then return end
   369→
   370→    -- Ensure fire mode is active
   371→    if self:GetNWInt("FireMode") == 0 then self:SetNWInt("FireMode", 1) end
   372→
   373→    -- Play third-person reload animation
   374→    self.Owner:SetAnimation(PLAYER_RELOAD)
   375→    self.NextReload = ct + 0.5
   376→
   377→    -- Call PreReload (sets b_clip for pump logic)
   378→    self:PreReload()
   379→
   380→    -- Handle flashlight/flare cancellation (server-side)
   381→    if SERVER then
   382→        if self.Owner:GetNWBool("UH_Flashlight") then
   383→            self.Owner:SetNWBool("UH_Flashlight", false)
   384→            self.Owner:SetNWBool("UH_ArmGone", false)
   385→            if not self.IsBolt and not self.IsPump and not self.TwoHanded then
   386→                self.Owner:SetNWFloat("UH_ArmTime", ct + 0.25)
   387→            end
   388→            self.Owner:EmitSound("uh/flashlight.wav")
   389→            net.Start("UH_Flashlight")
   390→                net.WriteEntity(self.Owner)
   391→                net.WriteBool(false)
   392→            net.Broadcast()
   393→        elseif self.Owner:GetNWBool("UH_Flare") then
   394→            UHThrowFlare(self.Owner)
   395→            self.Owner:SetNWBool("UH_ArmGone", false)
   396→            if not self.IsBolt and not self.IsPump and not self.TwoHanded then
   397→                self.Owner:SetNWFloat("UH_ArmTime", ct + 0.25)
   398→            end
   399→        end
   400→    end
   401→
   402→    -- Play the start reload animation
   403→    self:EasySendWeaponAnim("start_reload", ACT_SHOTGUN_RELOAD_START)
   404→    self:SetupAnimSounds("start_reload")
   405→
   406→    -- Get the duration of the start_reload animation
   407→    local vm = self.Owner:GetViewModel()
   408→    local startDur = IsValid(vm) and vm:SequenceDuration() or 0.5
   409→
   410→    -- Set up our own timer for shell insertion
   411→    -- First shell inserts RIGHT WHEN start_reload finishes (no extra gap).
   412→    local shellInterval = self.Primary.ReloadTime or 0.4
   413→    self._krm_nextShell = ct + startDur
   414→
   415→    -- Also set reloaddelay for compatibility with base class code
   416→    self.reloaddelay = self._krm_nextShell
   417→
   418→    -- Block firing during reload start
   419→    self:SetNextPrimaryFire(ct + startDur + 0.1)
   420→    self:SetNextSecondaryFire(ct + startDur + 0.1)
   421→    self.NextReload = ct + startDur + 0.1
   422→
   423→    -- Set reload state
   424→    self:SetUHBool("Reloading", true)
   425→    self:SetUHBool("Zooming", false)
   426→
   427→    -- Set up HUD reload timer
   428→    local num = math.min(self.Primary.ClipSize - self:Clip1(), self.Owner:GetAmmoCount(self:GetPrimaryAmmoType()))
   429→    local amount = num * shellInterval
   430→    self:SetNWFloat("ReloadTime", amount)
   431→    self:SetNWFloat("ReloadEndTime", ct + startDur + amount)
   432→end

   434→-- ============================================================
   435→-- RELOAD SHOTGUN (fully self-contained, timer-based)
   436→-- ============================================================
   437→function SWEP:ReloadShotgun(ct)
   438→    if not self:GetUHBool("Reloading") then return end
   439→
   440→    -- ==========================================
   441→    -- STOP CONDITIONS: clip full, no reserve, or fire pressed
   442→    -- ==========================================
   443→    if self:Clip1() >= self.Primary.ClipSize
   444→        or self.Owner:GetAmmoCount(self:GetPrimaryAmmoType()) <= 0
   445→        or self.Owner:KeyPressed(IN_ATTACK) then
   446→
   447→        self:ClearAnimSounds()
   448→
   449→        -- Play the finish animation
   450→        self:EasySendWeaponAnim("after_reload", ACT_SHOTGUN_RELOAD_FINISH)
   451→        self:SetupAnimSounds("after_reload")
   452→
   453→        local vm = self.Owner:GetViewModel()
   454→        local endDur = IsValid(vm) and vm:SequenceDuration() or 0.8
   455→
   456→        self:SetUHBool("Reloading", false)
   457→        self:SetNWFloat("ReloadTime", 0)
   458→        self:SetNWFloat("ReloadEndTime", 0)
   459→        self:SetNextPrimaryFire(ct + endDur)
   460→        self:SetNextSecondaryFire(ct + endDur)
   461→        self.NextReload = ct + endDur
   462→
   463→        self._krm_nextShell = nil
   464→        self.reloaddelay = nil
   465→
   466→        self:PostReload()
   467→        return
   468→    end
   469→
   470→    -- ==========================================
   471→    -- TIMER-BASED SHELL INSERTION
   472→    -- ==========================================
   473→    if self._krm_nextShell and ct >= self._krm_nextShell then
   474→        local shellInterval = self.Primary.ReloadTime or 0.4
   475→
   476→        -- Schedule the next shell insertion
   477→        self._krm_nextShell = ct + shellInterval
   478→
   479→        -- Play the per-shell reload animation
   480→        self:EasySendWeaponAnim("reload_loop", ACT_VM_RELOAD)
   481→        self:SetupAnimSounds("reload_loop")
   482→
   483→        -- Play the shell insertion sound (if set)
   484→        if self.Primary.ShellSound then
   485→            if (game.SinglePlayer() and SERVER) or (CLIENT and IsFirstTimePredicted()) then
   486→                self.Owner:EmitSound(self.Primary.ShellSound, 75, 100, 1, CHAN_USER_BASE)
   487→            end
   488→        end
   489→
   490→        -- Insert shell (server-authoritative)
   491→        if SERVER then
   492→            self:SetClip1(self:Clip1() + 1)
   493→            self.Owner:RemoveAmmo(1, self.Primary.Ammo, false)
   494→        end
   495→    end
   496→end

   508→-- Called when the BO3 parkour addon starts a vault or mantle.
   509→-- Plays the ["mantle"] viewmodel animation and blocks weapon actions.
   510→
   511→function SWEP:StartMantle()
   512→    local ct = CurTime()
   513→    local sp = game.SinglePlayer()
   514→    local iftp = IsFirstTimePredicted()
   515→
   516→    if not (sp or iftp) then return end
   517→
   518→    -- Cancel conflicting states
   519→    self:SetUHBool("Zooming", false)
   520→    if self:GetUHBool("Running") then
   521→        self:SetUHBool("Running", false)
   522→    end
   523→    if self._meleeActive then
   524→        self:EndMelee()
   525→    end
   526→
   527→    -- Cancel reload
   528→    if self:GetUHBool("Reloading") then
   529→        self:SetUHBool("Reloading", false)
   530→        self:SetNWFloat("ReloadTime", 0)
   531→        self:SetNWFloat("ReloadEndTime", 0)
   532→        if timer.Exists("UHReload_"..self.Owner:SteamID()) then
   533→            timer.Remove("UHReload_"..self.Owner:SteamID())
   534→        end
   535→    end
   536→
   537→    -- Set mantle state
   538→    self._mantleActive = true
   539→    self._engineWantsIdle = nil
   540→    self._customIdleActive = false
   541→    self:ClearAnimSounds()
   542→
   543→    -- Play ["mantle"] animation
   544→    self:EasySendWeaponAnim("mantle", ACT_VM_MELEE_SHOVE)
   545→
   546→    local vm = self.Owner:GetViewModel()
   547→    local animDuration = IsValid(vm) and vm:SequenceDuration() or (self.MantleDuration or 0.6)
   548→
   549→    -- Use whichever is longer: animation duration or addon traversal duration
   550→    local mantleDur = math.max(animDuration, self.MantleDuration or 0.6)
   551→    self._mantleEndTime = ct + mantleDur
   552→
   553→    -- Lock controls
   554→    self:SetNextPrimaryFire(ct + mantleDur)
   555→    self:SetNextSecondaryFire(ct + mantleDur)
   556→    self.NextReload = ct + mantleDur
   557→end

   572→function SWEP:Think()
   573→    local ct = CurTime()
   574→
   575→    -- ==========================================
   576→    -- PARKOUR TRAVERSAL DETECTION
   577→    -- ==========================================
   578→    -- Detects when the BO3 parkour addon sets BO3_IsVaulting / BO3_IsMantling.
   579→    local ply = self.Owner
   580→    if IsValid(ply) then
   581→        local isVaulting = ply:GetNW2Bool("BO3_IsVaulting", false)
   582→        local isMantling = ply:GetNW2Bool("BO3_IsMantling", false)
   583→        local isInTraversal = isVaulting or isMantling
   584→
   585→        if isInTraversal and not self._mantleActive then
   586→            if SERVER or IsFirstTimePredicted() then
   587→                self:StartMantle()
   588→            end
   589→        end
   590→
   591→        if not isInTraversal and self._mantleActive then
   592→            self:EndMantle()
   593→        end
   594→
   595→        self._lastBO3IsVaulting = isVaulting
   596→        self._lastBO3IsMantling = isMantling
   597→    end
   598→
   599→    -- Mantle timeout safety
   600→    if self._mantleActive and self._mantleEndTime and ct >= self._mantleEndTime then
   601→        self:EndMantle()
   602→    end
   603→
   604→    -- Melee hit timing
   605→    if self._meleeActive and not self._meleeHitDone and self._meleeHitTime and ct >= self._meleeHitTime then
   606→        self._meleeHitDone = true
   607→        if SERVER or IsFirstTimePredicted() then
   608→            self:DoMeleeTrace()
   609→        end
   610→    end
   611→
   612→    -- Melee end timing
   613→    if self._meleeActive and self._meleeEndTime and ct >= self._meleeEndTime then
   614→        self:EndMelee()
   615→    end
   616→
   617→    -- Holster finish
   618→    if self._holstering and self._holsterFinish and ct >= self._holsterFinish then
   619→        self._holstering = nil
   620→        self._holsterFinish = nil
   621→        if SERVER or IsFirstTimePredicted() then
   622→            local target = self._holsterTarget
   623→            self._holsterTarget = nil
   624→            if IsValid(target) then
   625→                self.Owner:SelectWeapon(target:GetClass())
   626→            end
   627→        end
   628→    end
   629→
   630→    -- ==========================================
   631→    -- RELOAD SHOTGUN — called directly, not through BaseClass.Think
   632→    -- ==========================================
   633→    if self:GetUHBool("Reloading") then
   634→        self:ReloadShotgun(ct)
   635→    end
   636→
   637→    BaseClass.Think(self)
   638→    self:HandleSprintingAnimations()
   639→end

   641→function SWEP:Holster()
   642→    if self._meleeActive then
   643→        self._meleeActive = false
   644→        self._meleeHitTime = nil
   645→        self._meleeHitDone = nil
   646→        self._meleeEndTime = nil
   647→        self:ClearAnimSounds()
   648→    end
   649→    if self._mantleActive then
   650→        self._mantleActive = false
   651→        self._mantleEndTime = nil
   652→        self:ClearAnimSounds()
   653→    end
   654→    -- Cancel any in-flight burst
   655→    self._burstRemaining = nil
   656→    self._burstNextFire = nil
   657→    -- Cancel rechamber
   658→    self._isRechambering = false
   659→    -- Reset animation transition trackers
   660→    self.wasZooming = false
   661→    self.wasRunning = false
   662→    -- Note: timer.Simple closures will still fire but the above
   663→    -- state resets make their checks self-correcting
   664→    return BaseClass.Holster(self)
   665→end

   702→function SWEP:HandleRunning( ct )
   703→    if self:GetUHBool("Reloading") then
   704→        return
   705→    end
   706→        
   707→        local vm = self.Owner:GetViewModel()
   708→    if IsValid(vm) then
   709→        local fireDelay = self:GetNextPrimaryFire() - ct
   710→        if fireDelay > 0.3 then return end
   711→    end
   712→        
   713→        if self:GetNWFloat("DeployTime", 10) then
   714→
   715→
   716→    local dist = self.Owner:GetVelocity():LengthSqr()
   717→    local isSprinting = self.Owner:KeyDown( IN_SPEED ) and dist > self.Owner:GetWalkSpeed()^2
   718→    local isSafeMode = self:GetNWInt("FireMode") == 0
   719→
   720→    -- 1. SPRINTING (Takes Priority)
   721→    if isSprinting then
   722→        self:SetHoldType( self.PassiveAnim )
   723→        self:SetUHBool("Running", true) -- Set to TRUE so animations play
   724→        self:SetUHBool("Zooming", false)
   725→        
   726→        if self:GetUHBool("Reloading") then
   727→            self:SetUHBool("Reloading", false)
   728→            self.NextReload = ct + 0.5
   729→            if timer.Exists("UHReload_"..self.Owner:SteamID()) then
   730→                timer.Remove( "UHReload_"..self.Owner:SteamID() )
   731→            end
   732→        end
   733→
   734→    -- 2. NOT SPRINTING
   735→                else
   736→                        self:SetHoldType( self.HoldType )
   737→        
   738→                        -- If we are in Safe Mode (and not sprinting), ensure we are flagged as NOT running.
   739→                        -- This allows the procedural lowering to take effect.
   740→                        if isSafeMode then
   741→                                self:SetUHBool("Running", false) 
   742→                        else
   743→                                self:SetUHBool("Running", false)
   744→                        end
   745→                end
   746→        end
   747→end

   749→function SWEP:GetViewModelPosition(pos, ang)
   750→    local ft = FrameTime()
   751→
   752→    -- Pass through BaseClass logic (Sway, Inspect, etc.)
   753→    if BaseClass and BaseClass.GetViewModelPosition then
   754→        pos, ang = BaseClass.GetViewModelPosition(self, pos, ang)
   755→    end
   756→
   757→    -- ==========================================
   758→    -- HYBRID LOWERING LOGIC (Safe Mode)
   759→    -- ==========================================
   760→    
   761→    -- We only use procedural LoweredPos/Ang if we are in SAFE MODE
   762→    -- AND we are NOT Sprinting.
   763→    local target = 0
   764→    if self:GetNWInt("FireMode") == 0 and not self:GetUHBool("Running") then
   765→        target = 1
   766→    end
   767→
   768→    self._uhLower = Lerp(ft * 8, self._uhLower or 0, target)
   769→
   770→    -- Apply Safe Mode Lowering (Procedural)
   771→    if self._uhLower > 0.001 then
   772→        local lp = self.LoweredPos or vector_origin
   773→        local la = self.LoweredAng or angle_zero
   774→
   775→        local ap, ay, ar = 0, 0, 0
   776→        if isangle(la) then
   777→            ap, ay, ar = la.p, la.y, la.r
   778→        elseif isvector(la) then
   779→            ap, ay, ar = la.x, la.y, la.z
   780→        end
   781→
   782→        ang:RotateAroundAxis(ang:Right(),   ap * self._uhLower)
   783→        ang:RotateAroundAxis(ang:Up(),      ay * self._uhLower)
   784→        ang:RotateAroundAxis(ang:Forward(), ar * self._uhLower)
   785→
   786→        pos = pos
   787→            + ang:Right()   * lp.x * self._uhLower
   788→            + ang:Forward() * lp.y * self._uhLower
   789→            + ang:Up()      * lp.z * self._uhLower
   790→    end
   791→
   792→    -- ==========================================
   793→    -- ALTERNATIVE POS/ANG LOGIC (Ultra Guns Style)
   794→    -- ==========================================
   795→    
   796→    -- Determine if we should show the offset
   797→    -- We hide it if: Sprinting, Zooming, or in Safe Mode
   798→    local targetAlt = 1
   799→    if self:GetUHBool("Running") or self:GetUHBool("Zooming") or self:GetNWInt("FireMode") == 0 then
   800→        targetAlt = 0
   801→    end
   802→
   803→    -- Smoothly interpolate the factor
   804→    self._altFactor = Lerp(ft * 10, self._altFactor or 0, targetAlt)
   805→
   806→    -- Apply the offset if it exists
   807→    if (self.AlternativePos or self.AlternativeAng) and self._altFactor > 0.01 then
   808→        local ap = self.AlternativePos or vector_origin
   809→        local aa = self.AlternativeAng or angle_zero
   810→
   811→        -- Apply Position Offset
   812→        pos = pos
   813→            + ang:Right()   * ap.x * self._altFactor
   814→            + ang:Forward() * ap.y * self._altFactor
   815→            + ang:Up()      * ap.z * self._altFactor
   816→
   817→        -- Apply Angle Offset
   818→        local ap_p, ap_y, ap_r = 0, 0, 0
   819→        if isangle(aa) then
   820→            ap_p, ap_y, ap_r = aa.p, aa.y, aa.r
   821→        elseif isvector(aa) then
   822→            ap_p, ap_y, ap_r = aa.x, aa.y, aa.z
   823→        end
   824→
   825→        ang:RotateAroundAxis(ang:Right(),   ap_p * self._altFactor)
   826→        ang:RotateAroundAxis(ang:Up(),      ap_y * self._altFactor)
   827→        ang:RotateAroundAxis(ang:Forward(), ap_r * self._altFactor)
   828→    end
   829→
   830→    -- ==========================================
   831→    -- SPRINT Z-AXIS OFFSET (Smoothed)
   832→    -- ==========================================
   833→    local targetZ = self:GetUHBool("Running") and 0 or 0
   834→    self._sprintZOffset = Lerp(ft * 10, self._sprintZOffset or 0, targetZ)
   835→    pos = pos + ang:Up() * self._sprintZOffset
   836→
   837→    return pos, ang
   838→end

   840→function SWEP:HandleSprintingAnimations()
   841→    local ply = self:GetOwner()
   842→    
   843→    if not IsValid(ply) or not ply:IsPlayer() then return end
   844→
   845→    -- Use UH variable for running state
   846→    local isRunning = self:GetUHBool("Running")
   847→    local isReloading = self:GetUHBool("Reloading")
   848→
   849→    -- Initialize tracking variable
   850→    if self.wasRunning == nil then self.wasRunning = false end
   851→
   852→    local vm = ply:GetViewModel()
   853→    if not IsValid(vm) then return end
   854→
   855→    -- TRANSITION: Enter Sprint
   856→    if isRunning and not self.wasRunning then
   857→        if not isReloading then
   858→            self:EasySendWeaponAnim("sprint_in", ACT_VM_SPRINT_ENTER)
   859→        end
   860→    
   861→    -- TRANSITION: Exit Sprint
   862→    elseif not isRunning and self.wasRunning then
   863→        if not isReloading then
   864→            self:EasySendWeaponAnim("sprint_out", ACT_VM_SPRINT_LEAVE)
   865→        end
   866→    
   867→    -- LOOPING: While Sprinting (and not reloading)
   868→    elseif isRunning and not isReloading then
   869→        -- If the current sprint animation finished (cycle >= 1), restart it
   870→        if vm:GetCycle() >= 1 then
   871→            self:EasySendWeaponAnim("sprint_idle", ACT_VM_SPRINT_IDLE)
   872→        end
   873→    end
   874→
   875→    self.wasRunning = isRunning
   876→end
