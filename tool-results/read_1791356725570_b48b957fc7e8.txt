   160→function SWEP:EasySendWeaponAnim(lookupanimation, elseifnotfounded)

   164→    if self.Animations and self.Animations[lookupanimation] then
   165→        local animData = self.Animations[lookupanimation]
   166→        if istable(animData) then
   167→            animData = animData[math.random(1, #animData)]
   168→        end
   169→        local seq = vm:LookupSequence(animData)
   170→        if seq and seq >= 0 then  -- FIXED: was > 0
   171→            self:SendSequence(vm, seq)
   172→        else
   173→            local act = type(elseifnotfounded) == "number" and vm:SelectWeightedSequence(elseifnotfounded) or -1
   174→            if act >= 0 then self:SendSequence(vm, act) end
   175→        end
   176→    else
   177→        local act = type(elseifnotfounded) == "number" and vm:SelectWeightedSequence(elseifnotfounded) or -1
   178→        if act >= 0 then self:SendSequence(vm, act) end

   203→-- ============================================================
   204→-- AnimSounds system
   205→-- ============================================================
   206→function SWEP:SetupAnimSounds(animKey, variantName)
   207→    if not self.AnimSounds then return end
   208→    local timeline = nil
   209→    if variantName and self.AnimSounds[variantName] then
   210→        timeline = self.AnimSounds[variantName]
   211→    end
   212→    if not istable(timeline) or #timeline == 0 then
   213→        timeline = self.AnimSounds[animKey]
   214→    end
   215→    if (not timeline or not istable(timeline)) and animKey == "reload_empty" then
   216→        timeline = self.AnimSounds["reload"]
   217→    end
   218→    if not istable(timeline) or #timeline == 0 then return end
   219→
   220→    self._animSoundTimeline = timeline
   221→    self._animSoundIndex = 1
   222→    self._animSoundStartTime = CurTime()
   223→    local vm = IsValid(self.Owner) and self.Owner:GetViewModel() or nil
   224→    self._animSoundPlaybackRate = IsValid(vm) and vm:GetPlaybackRate() or 1
   225→    self._animSoundActive = true
   226→end
   227→
   228→function SWEP:ProcessAnimSounds()
   229→    if not self._animSoundActive then return end
   230→    local sp = game.SinglePlayer()
   231→    local iftp = IsFirstTimePredicted()
   232→    -- FIXED: standardized prediction gate — server always runs for MP broadcast
   233→    if not (SERVER or (CLIENT and iftp)) then return end
   234→
   235→    local owner = self:GetOwner()
   236→    if not IsValid(owner) then
   237→        self._animSoundActive = false
   238→        return
   239→    end
   240→
   241→    local timeline = self._animSoundTimeline
   242→    local idx = self._animSoundIndex
   243→    if not timeline or idx > #timeline then
   244→        self._animSoundActive = false
   245→        return
   246→    end
   247→
   248→    local elapsed = (CurTime() - self._animSoundStartTime) * (self._animSoundPlaybackRate or 1)
   249→    while idx <= #timeline and elapsed >= (timeline[idx].time or 0) do
   250→        local entry = timeline[idx]
   251→        if entry.sound and entry.sound ~= "" and entry.sound ~= "nil" then
   252→            local snd = entry.sound
   253→            if istable(snd) then snd = snd[math.random(1, #snd)] end
   254→            owner:EmitSound(snd, entry.level or 75, entry.pitch or 100, 1, CHAN_USER_BASE)
   255→        end
   256→        if entry.callback then entry.callback(self) end
   257→        idx = idx + 1
   258→    end
   259→    self._animSoundIndex = idx
   260→end
   261→
   262→function SWEP:ClearAnimSounds()
   263→    self._animSoundActive = false
   264→    self._animSoundTimeline = nil
   265→    self._animSoundIndex = 0
   266→end

   308→-- ============================================================
   309→function SWEP:PrimaryAttack()
   310→    if not self:CanPrimaryAttack() then return end
   311→    local sp = game.SinglePlayer()
   312→    local iftp = IsFirstTimePredicted()
   313→
   314→    -- Firemode-specific shoot behavior
   315→    if SERVER or iftp then
   316→        local mode = self.FireModes[self:GetNWInt("FireMode")]
   317→        if mode and mode.shoot then
   318→            local res = mode.shoot(self.Owner, self)
   319→            if res then return end
   320→        end
   321→
   322→        -- FIXED: local spread (was global)
   323→        local dmg = math.random(self.Primary.MinDamage, self.Primary.MaxDamage)
   324→        if self:GetNWBool("Silenced") then
   325→            dmg = math.Round(dmg * 0.95)
   326→        end
   327→        self:ShootBullets(self.Owner:GetShootPos(), self.Owner:GetAimVector(), dmg, self.Penetration or 2)
   328→    end
   329→
   330→    local recoil = util.SharedRandom("uh_recoil", self.Primary.MinRecoil, self.Primary.MaxRecoil) * (self:GetUHBool("Zooming") and 0.35 or 1)
   331→
   332→    -- FIXED: standardized prediction gate — SERVER runs for MP broadcast
   333→    if SERVER or (CLIENT and iftp) then
   334→        self:DoMuzzleFlash()
   335→        self:CreateSmoke(self:GetMuzzle(), self.Primary.Delay + (self.Primary.Automatic and 0.14 or 0.32))
   336→        if not self.NoShell then
   337→            self:CreateShell(self.ShellDelay or 0, self.ShellHeat)
   338→        end
   339→        self.Owner:SetEyeAngles(self.Owner:EyeAngles() + Angle(recoil, 0, 0))
   340→    end
   341→
   342→    self.Owner:ViewPunch(Angle(recoil, 0, 0))
   343→
   344→    -- Animation
   345→    self:SendWeaponAnim(ACT_VM_IDLE)
   346→    local baseAct = ACT_VM_PRIMARYATTACK
   347→    if self.ShootAnimation then
   348→        baseAct = self:ShootAnimation()
   349→    end
   350→    local isSilenced = self:GetNWBool("Silenced")
   351→    local isZooming = self:GetUHBool("Zooming")
   352→    local lookupKey = isZooming and "shoot_ads" or "shoot"
   353→    if isSilenced then lookupKey = lookupKey .. "_sil" end
   354→    if not self.Animations or not self.Animations[lookupKey] then
   355→        lookupKey = tostring(baseAct)
   356→        if isSilenced then lookupKey = lookupKey .. "_sil" end
   357→    end
   358→
   359→    local fallbackActivity = baseAct
   360→    if isSilenced then
   361→        if baseAct == ACT_VM_PRIMARYATTACK then
   362→            fallbackActivity = ACT_VM_PRIMARYATTACK_SILENCED or ACT_VM_PRIMARYATTACK
   363→        elseif baseAct == ACT_VM_SECONDARYATTACK then
   364→            fallbackActivity = ACT_VM_SECONDARYATTACK_SILENCED or ACT_VM_SECONDARYATTACK
   365→        end
   366→    end
   367→
   368→    self:EasySendWeaponAnim(lookupKey, fallbackActivity)
   369→
   370→    local vm = IsValid(self.Owner) and self.Owner:GetViewModel() or nil
   371→    if IsValid(vm) then
   372→        vm:SetCycle(0)
   373→        vm:SetPlaybackRate(1)
   374→    end
   375→
   376→    self.Owner:SetAnimation(PLAYER_ATTACK1)
   377→    self.Owner:MuzzleFlash()
   378→
   379→    local fireSound = self:GetShootSound()
   380→    self:EmitSound(fireSound, 110, 100, 1, CHAN_WEAPON)
   381→
   382→    self:SetNextPrimaryFire(CurTime() + self.Primary.Delay)
   383→    self:SetNextSecondaryFire(CurTime() + self.Primary.Delay)
   384→    self:TakePrimaryAmmo(self.Primary.TakeAmmo)
   385→    self.NextReload = CurTime() + 0.5
   386→    self:PostShoot()
   387→end

   390→-- ShootBullets — FIXED: local spread (was global)
   391→-- ============================================================
   392→function SWEP:ShootBullets(pos, buldir, dmg, tries)
   393→    if tries <= 0 then return end
   394→    local owner = self.Owner
   395→
   396→    -- FIXED: local spread (was undeclared global)
   397→    local spread
   398→    local movement = Vector(owner:GetVelocity().x, owner:GetVelocity().y, 0):LengthSqr()
   399→    local movepercent = math.Clamp(movement / owner:GetRunSpeed()^2, 0, 1)
   400→    local move = movepercent * 0.1
   401→
   402→    if owner:OnGround() then
   403→        if self:GetUHBool("Zooming") and owner:Crouching() then
   404→            spread = self.Primary.Spread * 0.18
   405→        elseif self:GetUHBool("Zooming") then
   406→            spread = self.Primary.Spread * 0.26
   407→        elseif owner:Crouching() then
   408→            spread = self.Primary.Spread * 0.34
   409→        else
   410→            spread = self.Primary.Spread * 0.5
   411→        end
   412→    else
   413→        spread = self.Primary.Spread * 0.62
   414→    end
   415→    spread = spread + move
   416→
   417→    local bullet = {}
   418→    bullet.Num = self.Primary.NumberofShots
   419→    bullet.Src = pos
   420→    bullet.Dir = buldir
   421→    bullet.Spread = Vector(spread, spread, 0)
   422→    bullet.Tracer = 1
   423→    bullet.TracerName = "uh_tracer"
   424→    bullet.Force = self.Primary.Force
   425→    bullet.Damage = dmg
   426→    bullet.AmmoType = self.Primary.Ammo
   427→    bullet.Callback = function(attacker, bultrace, dmginfo)
   428→        local mat = bultrace.MatType
   429→        if SERVER then
   430→            util.ScreenShake(bultrace.HitPos, 5, 0.1, 0.5, 64)
   431→        end
   432→        if bultrace.Hit then
   433→            local dir = bultrace.HitNormal
   434→            local tr = {}
   435→            tr.start = bultrace.HitPos - dir * (self.PenetrationDepth or 4)
   436→            tr.endpos = bultrace.HitPos
   437→            tr.filter = owner
   438→            tr.mask = MASK_SHOT
   439→            local trace = util.TraceLine(tr)
   440→            if not trace.AllSolid and trace.Fraction > 0 and GetConVar("uh_sv_penetration"):GetBool() then
   441→                self:ShootBullets(trace.HitPos, buldir, math.Round(dmg * math.Rand(0.5, 0.6)), tries - 1)
   442→            end
   443→            if mat == MAT_METAL or mat == MAT_VENT or mat == MAT_GRATE then
   444→                local fx = EffectData()
   445→                fx:SetOrigin(bultrace.HitPos)
   446→                fx:SetScale(1)
   447→                util.Effect("uh_hitworld", fx)
   448→            end
   449→        end
   450→    end
   451→
   452→    owner:FireBullets(bullet)
   453→end

   481→function SWEP:CanPrimaryAttack()
   482→    if not IsValid(self.Owner) then return false end
   483→    if self:GetNWInt("FireMode") == 0
   484→       or self:GetNWFloat("DeployTime") > CurTime()
   485→       or self:GetUHBool("Running")
   486→       or self:GetUHBool("Reloading") then return false end
   487→    if self:Clip1() <= 0 then
   488→        if not self:GetUHBool("Reloading") then
   489→            self:EmitSound("Weapon_SMG1.Empty", 75, 100, 1, CHAN_USER_BASE)
   490→        end
   491→        self:SetNextPrimaryFire(CurTime() + 0.4)
   492→        return false
   493→    end
   494→    return true
   495→end

   513→            local mode = self:GetNWInt("FireMode") + 1
   514→            if mode > #self.FireModes then mode = 0 end
   515→
   516→            local old = self.FireModes[self:GetNWInt("FireMode")]
   517→            if old and old.holster then
   518→                old.holster(self.Owner, self)
   519→                if sp and SERVER then
   520→                    net.Start("UH_Select_Fire")
   521→                        net.WriteFloat(self:GetNWInt("FireMode"))
   522→                        net.WriteBool(false)
   523→                    net.Broadcast()
   524→                end

   526→                self:SendWeaponAnim(ACT_VM_LOWERED_TO_IDLE)
   527→            end
   528→
   529→            local new = self.FireModes[mode]
   530→            if new and new.equip then
   531→                new.equip(self.Owner, self)
   532→                if sp and SERVER then
   533→                    net.Start("UH_Select_Fire")
   534→                        net.WriteFloat(mode)
   535→                        net.WriteBool(true)
   536→                    net.Broadcast()
   537→                end
   538→            elseif mode == 0 then
   539→                self:SetUHBool("Running", false)
   540→                self:SetUHBool("Zooming", false)
   541→                self:SendWeaponAnim(ACT_VM_IDLE_TO_LOWERED)
   542→            end
   543→
   544→            self:SetNextPrimaryFire(ct + 0.5)
   545→            self:SetNextSecondaryFire(ct + 0.2)
   546→            self:SetNWInt("FireMode", mode)
   547→
   548→            -- FIXED: standardized sound gate
   549→            if SERVER or (CLIENT and IsFirstTimePredicted()) then
   550→                self.Owner:EmitSound("uh/flashlight.wav", 64, 100, 1, CHAN_USER_BASE)
   551→            end
   552→        end
   553→    end

   561→function SWEP:DoMuzzleFlash()

   693→function SWEP:Reload()
   694→    local ct = CurTime()
   695→    if not IsValid(self.Owner) then return end
   696→
   697→    -- FIXED: UH_GrenadeTime (was GrenadeTime)
   698→    if self.NextReload < ct and not self:GetUHBool("Running") and not self:GetUHBool("Reloading")
   699→       and not self.Owner:KeyDown(IN_USE)
   700→       and self.Owner:GetNWFloat("UH_GrenadeTime") < ct
   701→       and self:GetNWFloat("DeployTime") < ct then
   702→
   703→        if self:Clip1() < (self.Chambering and self.Primary.ClipSize + 1 or self.Primary.ClipSize)
   704→           and self.Owner:GetAmmoCount(self.Primary.Ammo) > 0 then
   705→
   706→            if SERVER then
   707→                self.Owner:SetAnimation(PLAYER_RELOAD)
   708→            end
   709→            if self:GetNWInt("FireMode") == 0 then self:SetNWInt("FireMode", 1) end
   710→
   711→            self:PreReload()
   712→
   713→            -- Play reload animation
   714→            local vm = IsValid(self.Owner) and self.Owner:GetViewModel() or nil
   715→            local reloadAnimKey = nil
   716→            local pickedVariant = nil
   717→
   718→            if self.ReloadAnim and type(self.ReloadAnim) == "string" then
   719→                local seq = IsValid(vm) and vm:LookupSequence(self.ReloadAnim) or -1
   720→                if seq and seq >= 0 then
   721→                    self:SendSequence(vm, seq)
   722→                end
   723→            else
   724→                local isSilenced = self:GetNWBool("Silenced")
   725→                local isEmpty = (self:Clip1() <= 0)
   726→                local animToPlay = nil
   727→
   728→                local function GetBestAnim(emptyKey, normalKey)
   729→                    local result
   730→                    if isSilenced then
   731→                        result = self.Animations[emptyKey.."_sil"] or self.Animations[emptyKey]
   732→                            or self.Animations[normalKey.."_sil"] or self.Animations[normalKey]
   733→                    else
   734→                        result = self.Animations[emptyKey] or self.Animations[normalKey]
   735→                    end
   736→                    if istable(result) then
   737→                        pickedVariant = result[math.random(1, #result)]
   738→                        result = pickedVariant
   739→                    end
   740→                    return result
   741→                end
   742→
   743→                if isEmpty then
   744→                    animToPlay = GetBestAnim("reload_empty", "reload")
   745→                    reloadAnimKey = "reload_empty"
   746→                else
   747→                    animToPlay = GetBestAnim("reload", "reload")
   748→                    reloadAnimKey = "reload"
   749→                end
   750→
   751→                if animToPlay then
   752→                    self:SendAnim(vm, animToPlay, ACT_VM_RELOAD)
   753→                else
   754→                    if self:GetNWBool("Silenced") then
   755→                        self:SendSequence(vm, vm:SelectWeightedSequence(ACT_VM_RELOAD_SILENCED or ACT_VM_RELOAD))
   756→                    else
   757→                        self:SendSequence(vm, vm:SelectWeightedSequence(ACT_VM_RELOAD))
   758→                    end
   759→                end
   760→            end
   761→
   762→            -- Reload duration
   763→            local AnimTime = self.ReloadTime or (IsValid(vm) and vm:SequenceDuration() or 2)
   764→            local speed = self.ReloadSpeed or 1
   765→            if speed ~= 1 then
   766→                AnimTime = AnimTime / speed
   767→                if IsValid(vm) then vm:SetPlaybackRate(speed) end
   768→            else
   769→                if IsValid(vm) then vm:SetPlaybackRate(1) end
   770→            end
   771→
   772→            self._reloadFinished = false
   773→            self._reloadEndTime = ct + AnimTime
   774→
   775→            -- Setup AnimSounds
   776→            if reloadAnimKey then
   777→                self:SetupAnimSounds(reloadAnimKey, pickedVariant)
   778→            end
   779→
   780→            self.NextReload = ct + AnimTime + 0.5
   781→            self:SetNextPrimaryFire(ct + AnimTime)
   782→            self:SetUHBool("Reloading", true)
   783→            self:SetUHBool("Zooming", false)
   784→            self:SetNWFloat("ReloadTime", AnimTime)
   785→            self:SetNWFloat("ReloadEndTime", ct + AnimTime)
   786→
   787→            -- ReloadTable sounds (if no AnimSounds entry exists)
   788→            if self.UseReloadTable and not self._animSoundActive then
   789→                local sp2 = game.SinglePlayer()
   790→                local iftp = IsFirstTimePredicted()
   791→                if (sp2 and SERVER) or (not sp2 and CLIENT and iftp) then
   792→                    local scale = 1 / speed
   793→                    for _, tbl in ipairs(self.ReloadTable) do
   794→                        if tbl and tbl.delay then
   795→                            -- FIXED: was `return` (exited entire Reload function)
   796→                            -- Now uses proper continue pattern
   797→                            timer.Simple(tbl.delay * scale, function()
   798→                                if not IsValid(self) or not self:GetUHBool("Reloading") then return end
   799→                                local owner = self:GetOwner()
   800→                                if not IsValid(owner) or owner:GetActiveWeapon() != self then return end
   801→                                if tbl.sound and tbl.sound ~= "" and tbl.sound ~= "nil" then
   802→                                    local snd = tbl.sound
   803→                                    if istable(snd) then snd = snd[math.random(1, #snd)] end
   804→                                    owner:EmitSound(snd, tbl.level or 75, tbl.pitch or 100, 1, CHAN_USER_BASE)
   805→                                end
   806→                            end)
   807→                        end
   808→                    end
   809→                end
   810→            end
   811→        end
   812→    end
   813→end

   861→function SWEP:Think()
   862→    BaseClass.Think(self)
   863→    if not IsValid(self.Owner) then return end
   864→    local ct = CurTime()

   866→    -- Reload speed enforcement + completion

   882→        or self.Owner:GetNWBool("UH_Flare")
   883→        or self.Owner:GetNWBool("UH_Flashlight")
   884→        or self:GetUHBool("Running")
   885→        or self:GetUHBool("Reloading")
   886→        or self:GetNWFloat("DeployTime") > ct
   887→        or self.Owner:GetNWFloat("UH_GrenadeTime") > ct
   888→
   889→    local wantsToZoom = self.Owner:KeyDown(IN_ATTACK2)
   890→        and not preventZoom
   891→        and not self.Owner:KeyDown(IN_USE)
   892→        and self:GetNWInt("FireMode") ~= 0
   893→
   894→    if isZooming and preventZoom then
   895→        self:SetUHBool("Zooming", false)
   896→        if SERVER or (CLIENT and IsFirstTimePredicted()) then
   897→            self.Owner:EmitSound("weapons/underhell/ironsight_off.wav", 65, 100, 1, CHAN_USER_BASE)
   898→        end
   899→    elseif isZooming and not wantsToZoom then
   900→        self:SetUHBool("Zooming", false)
   901→        if SERVER or (CLIENT and IsFirstTimePredicted()) then
   902→            self.Owner:EmitSound("weapons/underhell/ironsight_off.wav", 65, 100, 1, CHAN_USER_BASE)
   903→        end
   904→    elseif not isZooming and wantsToZoom then
   905→        self:SetUHBool("Zooming", true)
   906→        if SERVER or (CLIENT and IsFirstTimePredicted()) then
   907→            self.Owner:EmitSound("weapons/underhell/ironsight_on.wav", 64, 100, 1, CHAN_USER_BASE)
   908→        end
   909→    end
   910→
   911→    -- AnimSounds processing
   912→    self:ProcessAnimSounds()
   913→
   914→    if self.CustomThink then
   915→        self:CustomThink(ct)
   916→    end
