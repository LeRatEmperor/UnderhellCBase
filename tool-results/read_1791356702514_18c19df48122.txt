    34→-- ============================================================
    35→-- DEFAULT ATTACHMENT SYSTEM
    36→-- ============================================================
    37→-- A slot can define a `default` index. When no attachment is
    38→-- selected (sel == nil), the default is auto-applied.
    39→-- Use forceDefault = true to prevent selecting "None".
    40→--
    41→--   SWEP.Attachments = {
    42→--       [1] = { atts = { "opt_red", "opt_acog" }, default = 1 },
    43→--       [2] = { atts = { "mag_fast" } },  -- no default
    44→--       [3] = { atts = { "grip_v" }, default = 1, forceDefault = true },
    45→--   }
    46→-- ============================================================
    47→
    48→function SWEP:GetEffectiveAttachment(slot)
    49→    if not self.Attachments or not self.Attachments[slot] then return 0 end
    50→    local slotData = self.Attachments[slot]
    51→    local sel = slotData.sel
    52→
    53→    if sel == nil then
    54→        if slotData.default and slotData.default > 0 then
    55→            return slotData.default
    56→        end
    57→        return 0
    58→    end
    59→
    60→    if sel == 0 then
    61→        if slotData.forceDefault and slotData.default and slotData.default > 0 then
    62→            return slotData.default
    63→        end
    64→        return 0
    65→    end
    66→
    67→    return sel
    68→end

    92→-- ============================================================
    93→-- CAMERA BONE SYSTEM (BO3-style procedural camera animation)
    94→-- ============================================================
    95→-- This is a direct port of the proven working BO3 base implementation.
    96→-- Weapons declare:
    97→--   SWEP.CameraAttachment = "Camera"        -- $attachment name on VM
    98→--   SWEP.CameraReserve    = false            -- invert the angle
    99→--   SWEP.CameraOffset     = Angle(0,0,0)     -- extra angle offset
   100→--
   101→-- When CameraAttachment is nil/empty, the system does nothing.
   102→--
   103→-- ConVar: cl_cuh_camera_scale (default 1.0) — player-adjustable
   104→-- intensity multiplier. 0 = off, 1 = full BO3 intensity.
   105→-- ============================================================
   106→if CLIENT then
   107→    CUH_CAMERA_SCALE = CreateClientConVar("cl_cuh_camera_scale", "1.0", true, false,
   108→        "CUH camera bone animation scale (0 = off, 1 = full BO3 intensity)")
   109→end
   110→
   111→function SWEP:CalcView(ply, pos, ang, fov)
   112→    -- Camera bone angle tracking — applied to the PLAYER'S VIEW
   113→    if self.CameraAttachment and self.CameraAttachment ~= "" then
   114→        local vm = IsValid(self.Owner) and self.Owner:GetViewModel() or nil
   115→        if IsValid(vm) then
   116→            local seq = self.m_CurrentSequence or vm:GetSequenceName(vm:GetSequence()) or ""
   117→            if not string.find(seq, "Fire") and not string.find(seq, "Idle") then
   118→                local attID = vm:LookupAttachment(self.CameraAttachment)
   119→                if attID and attID > 0 then
   120→                    local att = vm:GetAttachment(attID)
   121→                    if att then
   122→                        if self.CameraOffset then
   123→                            ang:Add(self.CameraOffset)
   124→                        end
   125→                        local localAng = vm:WorldToLocalAngles(att.Ang)
   126→                        if self.CameraReserve then
   127→                            localAng:Mul(-1)
   128→                        end
   129→                        local scale = (CLIENT and CUH_CAMERA_SCALE and CUH_CAMERA_SCALE:GetFloat()) or 1
   130→                        localAng:Mul(scale)
   131→                        ang:Add(localAng)
   132→                    end
   133→                end
   134→            end
   135→        end
   136→    end
   137→
   138→    -- Pass to parent base for view bobbing and FOV zoom
   139→    if BaseClass.CalcView then
   140→        return BaseClass.CalcView(self, ply, pos, ang, fov)
   141→    end
   142→    return pos, ang, fov
   143→end

   152→-- ============================================================
   153→-- Stat cache is initialized lazily on first access (GetStat/SetStat)
   154→-- OR when ApplyAttachments runs. Both paths funnel through InitStatCache.
   155→--
   156→-- IMPORTANT: the per-instance table.Copy(self.Primary) MUST happen inside
   157→-- InitStatCache (not in a separate InitStatCacheIfNeeded helper). The
   158→-- previous design had the copy in a separate helper that ApplyAttachments
   159→-- bypassed by calling self:InitStatCache() directly — that left
   160→-- self.Primary pointing at the CLASS table, so attachment SetStat calls
   161→-- mutated the class table and bled state across every weapon of the
   162→-- same class (visible as fire-delay / fire-timing "stutter" because
   163→-- Primary.Delay was being clobbered on the shared class table).
   164→-- ============================================================
   165→function SWEP:InitStatCache()
   166→    if self._statCache then return end
   167→
   168→    -- Copy Primary/Secondary per-instance so attachment stat
   169→    -- modifications don't bleed across weapons of the same class.
   170→    -- Done here (not in Initialize) so the parent base's init still
   171→    -- sees the original shared class table during its own setup.
   172→    self.Primary   = table.Copy(self.Primary   or {})
   173→    self.Secondary = table.Copy(self.Secondary or {})
   174→
   175→    -- Copy Animations per-instance AND snapshot the origin so attachments
   176→    -- can override anim keys (e.g. magazine swapping reload -> reload_ext02)
   177→    -- and we can restore them on detach.
   178→    self.Animations = table.Copy(self.Animations or {})
   179→    self._animationsOrigin = table.Copy(self.Animations)
   180→
   181→    -- AnimSounds likewise — mag swaps may want different reload sounds.
   182→    if self.AnimSounds then
   183→        self.AnimSounds = table.Copy(self.AnimSounds)
   184→        self._animSoundsOrigin = table.Copy(self.AnimSounds)
   185→    end
   186→
   187→    self._statCache = {}
   188→    self._statOrigins = {}
   189→
   190→    -- Cache Primary table
   191→    for k, v in pairs(self.Primary or {}) do
   192→        self._statCache["Primary." .. k] = v
   193→        self._statOrigins["Primary." .. k] = v
   194→    end
   195→
   196→    -- Cache Secondary table
   197→    for k, v in pairs(self.Secondary or {}) do
   198→        self._statCache["Secondary." .. k] = v
   199→        self._statOrigins["Secondary." .. k] = v
   200→    end
   201→
   202→    -- Cache top-level fields (NOT Primary.* or Secondary.* —
   203→    -- those are already cached by the loops above)
   204→    local topLevel = {
   205→        "DamageGeneric", "Num", "FireRate", "Spread", "SpreadIronsighted",
   206→        "ViewSlideRecoilUp", "ViewSlideRecoilRight", "ZoomFov",
   207→        "IronsightSpeed", "IronsightEaseIn", "ZoomSpeedIn", "ZoomSpeedOut",
   208→        "IronSightTime", "MoveSpeed", "IronRecoilMultiplier",
   209→    }
   210→    for _, key in ipairs(topLevel) do
   211→        if string.find(key, "%.") then
   212→            local tbl, field = string.match(key, "(.+)%.(.+)")
   213→            if self[tbl] and self[tbl][field] ~= nil then
   214→                self._statCache[key] = self[tbl][field]
   215→                self._statOrigins[key] = self[tbl][field]
   216→            end
   217→        else
   218→            if self[key] ~= nil then
   219→                self._statCache[key] = self[key]
   220→                self._statOrigins[key] = self[key]
   221→            end
   222→        end
   223→    end
   224→
   225→    -- Cache IronSights positions
   226→    if self.IronSightsPos then
   227→        self._statCache["IronSightsPos"] = self.IronSightsPos
   228→        self._statOrigins["IronSightsPos"] = self.IronSightsPos
   229→    end
   230→    if self.IronSightsAng then
   231→        self._statCache["IronSightsAng"] = self.IronSightsAng
   232→        self._statOrigins["IronSightsAng"] = self.IronSightsAng
   233→    end
   234→
   235→    self._statDirty = false
   236→end
   237→
   238→function SWEP:GetStat(path)
   239→    if not self._statCache then self:InitStatCache() end
   240→    return self._statCache[path]
   241→end
   242→
   243→function SWEP:SetStat(path, value)
   244→    if not self._statCache then self:InitStatCache() end
   245→    self._statCache[path] = value
   246→    -- Eager write: immediately update the actual SWEP field
   247→    if string.find(path, "%.") then
   248→        local tbl, field = string.match(path, "(.+)%.(.+)")
   249→        if self[tbl] then self[tbl][field] = value end
   250→    else
   251→        self[path] = value
   252→    end
   253→end
   254→
   255→function SWEP:RestoreStat(path)
   256→    if not self._statOrigins then return end
   257→    local origin = self._statOrigins[path]
   258→    if origin ~= nil then self:SetStat(path, origin) end
   259→end
   260→
   261→-- Write cached stats back to actual SWEP fields
   262→function SWEP:FlushStats()
   263→    if not self._statCache or not self._statDirty then return end
   264→
   265→    for path, value in pairs(self._statCache) do
   266→        if string.find(path, "%.") then
   267→            local tbl, field = string.match(path, "(.+)%.(.+)")
   268→            if self[tbl] then self[tbl][field] = value end
   269→        else
   270→            self[path] = value
   271→        end
   272→    end
   273→
   274→    self._statDirty = false
   275→end

   278→-- VELEMENT SYSTEM — Viewmodel attachment models

   280→
   281→function SWEP:InitVElements()
   282→    if not self.ViewModelElements then return end
   283→    if self._vElementsInit then return end
   284→    self._vElementsInit = true
   285→
   286→    if CLIENT then
   287→        for name, elem in pairs(self.ViewModelElements) do
   288→            if elem._defaultActive == nil then
   289→                elem._defaultActive = elem.active or false
   290→            end
   291→            -- Snapshot the full element state so ApplyAttachments
   292→            -- can restore model, bodygroups, skin, material, etc.
   293→            -- (not just the active flag) when an attachment is removed.
   294→            if elem._defaultSnapshot == nil then
   295→                local snap = table.Copy(elem)
   296→                snap._defaultSnapshot = nil
   297→                snap._csModel = nil
   298→                elem._defaultSnapshot = snap
   299→            end
   300→            if elem.type == "Model" and elem.model and elem.model ~= "" then
   301→                elem._csModel = ClientsideModel(elem.model, RENDERGROUP_VIEWMODEL)
   302→                if IsValid(elem._csModel) then
   303→                    elem._csModel:SetNoDraw(true)
   304→                    if elem.skin then elem._csModel:SetSkin(elem.skin) end
   305→                    if elem.bodygroups then
   306→                        for bg, val in pairs(elem.bodygroups) do
   307→                            elem._csModel:SetBodygroup(bg, val)
   308→                        end
   309→                    end
   310→                    if elem.material and elem.material ~= "" then
   311→                        elem._csModel:SetMaterial(elem.material)
   312→                    end
   313→                end
   314→            end
   315→        end
   316→    end
   317→end

   319→function SWEP:DrawVElements(vm)
   320→    if not self.ViewModelElements then return end
   321→    if not self._vElementsInit then self:InitVElements() end
   322→
   323→    -- CRITICAL: Apply viewmodel bodygroups EVERY FRAME.
   324→    -- The viewmodel (v_scotia.mdl) has the default barrel/stock/mag/pgrip
   325→    -- baked into its mesh. Without setting bodygroups to hide them, BOTH
   326→    -- the viewmodel's default parts AND the VElement attachment models
   327→    -- are visible at the same time — causing z-fighting/overlap that
   328→    -- makes it look like "nothing changed".
   329→    -- ApplyBodygroupsVM sets vm:SetBodygroup(bg, val) for each entry in
   330→    -- self.Bodygroups_V, hiding the default parts so only the VElement
   331→    -- attachment models show.
   332→    if self.ApplyBodygroupsVM then self:ApplyBodygroupsVM(vm) end
   333→
   334→    -- ONE-TIME diagnostic print (only prints once, then every 5 seconds)
   335→    -- so we can confirm DrawVElements IS being called every frame.
   336→    if not self._drawVElemLastPrint or CurTime() - self._drawVElemLastPrint > 5 then
   337→        self._drawVElemLastPrint = CurTime()
   338→        local activeCount = 0
   339→        local validCount = 0
   340→        for name, elem in pairs(self.ViewModelElements) do
   341→            if elem.active then
   342→                activeCount = activeCount + 1
   343→                if IsValid(elem._csModel) then
   344→                    validCount = validCount + 1
   345→                end
   346→            end
   347→        end
   348→        print("[CUH-DBG] DrawVElements running  vm=" .. tostring(vm) .. "  activeElems=" .. activeCount .. "  validCsModels=" .. validCount)
   349→    end
   350→
   351→    for name, elem in pairs(self.ViewModelElements) do
   352→        if not elem.active then continue end
   353→
   354→        if elem.type == "Model" and IsValid(elem._csModel) then
   355→            local model = elem._csModel
   356→
   357→            -- CRITICAL: SetModel must be called every frame in case the
   358→            -- model string was swapped by ApplyAttachments. The _csModel
   359→            -- was created with the OLD model; if we don't call SetModel
   360→            -- with the NEW string, the old model keeps drawing.
   361→            if elem.model and elem.model ~= "" then
   362→                model:SetModel(elem.model)
   363→            end
   364→
   365→            if elem.bonemerge then
   366→                model:SetParent(vm)
   367→                model:AddEffects(EF_BONEMERGE)
   368→                -- Apply pos offset even for bonemerged elements.
   369→                -- EF_BONEMERGE drives the bones, but we can still nudge
   370→                -- the entity origin to close small gaps between parts.
   371→                if elem.pos and (elem.pos.x ~= 0 or elem.pos.y ~= 0 or elem.pos.z ~= 0) then
   372→                    local vmPos, vmAng = vm:GetPos(), vm:GetAngles()
   373→                    local offset = vmPos
   374→                        + vmAng:Right()   * (elem.pos.x or 0)
   375→                        + vmAng:Forward() * (elem.pos.y or 0)
   376→                        + vmAng:Up()      * (elem.pos.z or 0)
   377→                    model:SetPos(offset)
   378→                end
   379→            else
   380→                model:SetParent(NULL)
   381→                model:RemoveEffects(EF_BONEMERGE)
   382→
   383→                -- Manual bone positioning (only when NOT bonemerged)
   384→                local boneId = vm:LookupBone(elem.bone or "")
   385→                if boneId then
   386→                    local bPos, bAng = vm:GetBonePosition(boneId)
   387→                    if bPos then
   388→                        local pos = bPos
   389→                        pos = pos + bAng:Right()   * (elem.pos and elem.pos.x or 0)
   390→                        pos = pos + bAng:Forward() * (elem.pos and elem.pos.y or 0)
   391→                        pos = pos + bAng:Up()      * (elem.pos and elem.pos.z or 0)
   392→                        local ang = Angle(bAng)
   393→                        if elem.ang then
   394→                            ang:RotateAroundAxis(ang:Right(),   elem.ang.p or 0)
   395→                            ang:RotateAroundAxis(ang:Up(),       elem.ang.y or 0)
   396→                            ang:RotateAroundAxis(ang:Forward(),  elem.ang.r or 0)
   397→                        end
   398→                        model:SetPos(pos)
   399→                        model:SetAngles(ang)
   400→                    end
   401→                end
   402→            end
   403→
   404→            local scale = elem.scale or Vector(1, 1, 1)
   405→            model:SetModelScale(scale.x, 0)
   406→            if elem.material and elem.material ~= "" then model:SetMaterial(elem.material) end
   407→            if elem.skin then model:SetSkin(elem.skin) end
   408→            if elem.bodygroups then
   409→                for bg, val in pairs(elem.bodygroups) do model:SetBodygroup(bg, val) end
   410→            end
   411→
   412→            if elem.surpresslightning then render.SuppressEngineLighting(true) end
   413→            model:DrawModel()
   414→            if elem.surpresslightning then render.SuppressEngineLighting(false) end
   415→
   416→        elseif elem.type == "Sprite" then
   417→            local boneId = vm:LookupBone(elem.bone or "")
   418→            if boneId then
   419→                local bPos, bAng = vm:GetBonePosition(boneId)
   420→                if bPos then
   421→                    local pos = bPos
   422→                    pos = pos + bAng:Right()   * (elem.pos and elem.pos.x or 0)
   423→                    pos = pos + bAng:Forward() * (elem.pos and elem.pos.y or 0)
   424→                    pos = pos + bAng:Up()      * (elem.pos and elem.pos.z or 0)
   425→                    if elem.material and elem.material ~= "" then
   426→                        render.SetMaterial(Material(elem.material))
   427→                        local size = elem.size or 4
   428→                        render.DrawSprite(pos, size, size, elem.color or Color(255,255,255,255))
   429→                    end
   430→                end
   431→            end
   432→        end
   433→    end
   434→end

   436→function SWEP:CleanupVElements()
   437→    if not self.ViewModelElements then return end
   438→    for name, elem in pairs(self.ViewModelElements) do
   439→        if IsValid(elem._csModel) then
   440→            elem._csModel:Remove()
   441→            elem._csModel = nil
   442→        end
   443→    end
   444→    self._vElementsInit = false
   445→end

   511→function SWEP:DrawWElements(owner)
   512→    if not self.WorldModelElements then return end
   513→    if not self._wElementsInit then self:InitWElements() end
   514→    if not IsValid(owner) then return end
   515→
   516→    for name, elem in pairs(self.WorldModelElements) do
   517→        if not elem.active then continue end
   518→
   519→        if elem.type == "Model" and IsValid(elem._csModel) then
   520→            local model = elem._csModel
   521→
   522→            -- SetModel every frame in case it was swapped
   523→            if elem.model and elem.model ~= "" then
   524→                model:SetModel(elem.model)
   525→            end
   526→
   527→            if elem.bonemerge then
   528→                -- Bonemerge: parent to the world model ClientsideModel (not self).
   529→                -- The ClientsideModel is positioned at the player's hand bone,
   530→                -- so bonemerged elements inherit the correct position and rotation.
   531→                local parent = IsValid(self._wmClientModel) and self._wmClientModel or self
   532→                if model:GetParent() ~= parent then
   533→                    model:SetParent(parent)
   534→                end
   535→                if not model:IsEffectActive(EF_BONEMERGE) then
   536→                    model:AddEffects(EF_BONEMERGE)
   537→                end
   538→                model:SetLocalPos(vector_origin)
   539→                model:SetLocalAngles(angle_zero)
   540→            else
   541→                -- Manual positioning (for non-bonemerged elements)
   542→                local boneid = owner:LookupBone("ValveBiped.Bip01_R_Hand")
   543→                if boneid then
   544→                    local matrix = owner:GetBoneMatrix(boneid)
   545→                    if matrix then
   546→                        local handPos = matrix:GetTranslation()
   547→                        local handAng = matrix:GetAngles()
   548→
   549→                        local pos = handPos
   550→                        if elem.pos then
   551→                            pos = pos + handAng:Right()   * elem.pos.x
   552→                            pos = pos + handAng:Forward() * elem.pos.y
   553→                            pos = pos + handAng:Up()      * elem.pos.z
   554→                        end
   555→
   556→                        local ang = Angle(handAng)
   557→                        if elem.ang then
   558→                            ang:RotateAroundAxis(ang:Right(),   elem.ang.p or 0)
   559→                            ang:RotateAroundAxis(ang:Up(),       elem.ang.y or 0)
   560→                            ang:RotateAroundAxis(ang:Forward(),  elem.ang.r or 0)
   561→                        end
   562→
   563→                        model:SetPos(pos)
   564→                        model:SetAngles(ang)
   565→                        model:SetParent(NULL)
   566→                        model:RemoveEffects(EF_BONEMERGE)
   567→                    end
   568→                end
   569→            end
   570→
   571→            local scale = elem.scale or Vector(1, 1, 1)
   572→            model:SetModelScale(scale.x, 0)
   573→
   574→            if elem.material and elem.material ~= "" then model:SetMaterial(elem.material) end
   575→            if elem.skin then model:SetSkin(elem.skin) end
   576→            if elem.bodygroups then
   577→                for bg, val in pairs(elem.bodygroups) do model:SetBodygroup(bg, val) end
   578→            end
   579→
   580→            model:DrawModel()
   581→        end
   582→    end
   583→end

   585→function SWEP:CleanupWElements()
   586→    if not self.WorldModelElements then return end
   587→    for name, elem in pairs(self.WorldModelElements) do
   588→        if IsValid(elem._csModel) then
   589→            elem._csModel:Remove()
   590→            elem._csModel = nil
   591→        end
   592→    end
   593→    self._wElementsInit = false
   594→end

   596→-- ============================================================
   597→-- DRAW WORLD MODEL — render weapon + WElements
   598→-- ============================================================
   599→-- Uses a ClientsideModel for the receiver (not self:DrawModel) so we
   600→-- can position it at the player's hand bone via LocalToWorld.
   601→-- The bonemerged attachment elements are parented to this ClientsideModel.
   602→--
   603→-- Per-weapon config:
   604→--   SWEP.WorldModelOffset = Vector(0, -2, -1)    -- local offset from hand bone
   605→--   SWEP.WorldModelAngle  = Angle(180, 90, 0)    -- local angle from hand bone
   606→--   SWEP.WorldModelSkin   = 1                     -- skin index (optional)
   607→-- ============================================================
   608→if CLIENT then
   609→    -- Create the world model ClientsideModel lazily per weapon instance.
   610→    -- We store it on the weapon table and create it on first DrawWorldModel.
   611→end
   612→
   613→function SWEP:DrawWorldModel()
   614→    local owner = self:GetOwner()
   615→
   616→    -- Ensure we have a ClientsideModel for the receiver
   617→    if CLIENT and not IsValid(self._wmClientModel) then
   618→        self._wmClientModel = ClientsideModel(self.WorldModel, RENDERGROUP_OPAQUE)
   619→        if IsValid(self._wmClientModel) then
   620→            self._wmClientModel:SetNoDraw(true)
   621→            if self.WorldModelSkin then
   622→                self._wmClientModel:SetSkin(self.WorldModelSkin)
   623→            end
   624→        end
   625→    end
   626→
   627→    if IsValid(owner) then
   628→        -- Position the receiver at the player's hand bone
   629→        local boneid = owner:LookupBone("ValveBiped.Bip01_R_Hand")
   630→        if boneid then
   631→            local matrix = owner:GetBoneMatrix(boneid)
   632→            if matrix then
   633→                local offsetVec = self.WorldModelOffset or Vector(0, -2, -1)
   634→                local offsetAng = self.WorldModelAngle or Angle(180, 90, 0)
   635→                local newPos, newAng = LocalToWorld(offsetVec, offsetAng, matrix:GetTranslation(), matrix:GetAngles())
   636→
   637→                if IsValid(self._wmClientModel) then
   638→                    self._wmClientModel:SetPos(newPos)
   639→                    self._wmClientModel:SetAngles(newAng)
   640→                    self._wmClientModel:SetupBones()
   641→
   642→                    -- Update the model in case it was changed
   643→                    if self._wmClientModel:GetModel() ~= self.WorldModel then
   644→                        self._wmClientModel:SetModel(self.WorldModel)
   645→                    end
   646→
   647→                    self._wmClientModel:DrawModel()
   648→                else
   649→                    self:DrawModel()
   650→                end
   651→            end
   652→        end
   653→
   654→        -- Draw attachment elements (bonemerged to the weapon entity)
   655→        self:DrawWElements(owner)
   656→    else
   657→        -- No owner — draw at weapon position (ground spawn)
   658→        if IsValid(self._wmClientModel) then
   659→            self._wmClientModel:SetPos(self:GetPos())
   660→            self._wmClientModel:SetAngles(self:GetAngles())
   661→            self._wmClientModel:DrawModel()
   662→        else
   663→            self:DrawModel()
   664→        end
   665→    end
   666→end

   729→-- ============================================================
   730→-- APPLY ATTACHMENTS — rebuild stats, VElements, WElements, bodygroups
   731→-- ============================================================
   732→
   733→function SWEP:ApplyAttachments()
   734→    print("[CUH-DBG] >>> ApplyAttachments CALLED  realm=" .. (SERVER and "SERVER" or "CLIENT")
   735→        .. "  wep=" .. tostring(self) .. "  class=" .. tostring(self.GetClass and self:GetClass() or "?"))
   736→    if not self.Attachments then
   737→        print("[CUH-DBG]   ABORT: self.Attachments is nil")
   738→        return
   739→    end
   740→    if not self._statOrigins then self:InitStatCache() end
   741→
   742→    print("[CUH-DBG]   CustomUH.Attachments count = " .. tostring(CustomUH and CustomUH.Attachments and table.Count(CustomUH.Attachments) or "nil"))
   743→    if CustomUH and CustomUH.Attachments then
   744→        local keys = {}
   745→        for k, _ in pairs(CustomUH.Attachments) do keys[#keys+1] = k end
   746→        print("[CUH-DBG]   CustomUH.Attachments keys = {" .. table.concat(keys, ", ") .. "}")
   747→    end
   748→
   749→    -- 0. ENSURE SNAPSHOT — call InitVElements/InitWElements once now, BEFORE
   750→    -- any attachment modifies the model. This guarantees the _defaultSnapshot
   751→    -- captures the DEFAULT model state (not a previously-attached model).
   752→    --
   753→    -- Why this matters: InitVElements lazily snapshots the elem on first
   754→    -- call. If the first call happens to be AFTER an attachment's Attach()
   755→    -- has already swapped elem.model, the snapshot would capture the swapped
   756→    -- model — and Step 2 below would "restore" to the wrong (already-swapped)
   757→    -- model on every subsequent ApplyAttachments call. Calling Init here
   758→    -- (before Step 5's Attach calls) prevents that.
   759→    --
   760→    -- InitVElements/InitWElements are idempotent (guarded by _vElementsInit
   761→    -- / _wElementsInit), so this is a no-op if PostDrawViewModel already
   762→    -- ran the snapshot.
   763→    if self.InitVElements then self:InitVElements() end
   764→    if self.InitWElements then self:InitWElements() end
   765→
   766→    -- 1. FULL RESET: restore all stats to origin values
   767→    for path, _ in pairs(self._statCache) do
   768→        self:RestoreStat(path)
   769→    end
   770→
   771→    -- 1b. RESET ANIMATIONS to origin (per-instance snapshot taken in InitStatCache).
   772→    -- Attachments can override anim keys (e.g. magazine swapping
   773→    -- reload -> reload_ext02). Restore them before re-applying.
   774→    if self._animationsOrigin then
   775→        self.Animations = table.Copy(self._animationsOrigin)
   776→    end
   777→    if self._animSoundsOrigin then
   778→        self.AnimSounds = table.Copy(self._animSoundsOrigin)
   779→    end
   780→
   781→    -- 2. RESET VELEMENTS to default state (full restore from snapshot)
   782→    if self.ViewModelElements then
   783→        for name, elem in pairs(self.ViewModelElements) do
   784→            if elem._defaultSnapshot then
   785→                for k, v in pairs(elem._defaultSnapshot) do
   786→                    elem[k] = v
   787→                end
   788→            else
   789→                elem.active = elem._defaultActive or false
   790→            end
   791→        end
   792→    end
   793→
   794→    -- 3. RESET WELEMENTS to default state (full restore from snapshot)
   795→    if self.WorldModelElements then
   796→        for name, elem in pairs(self.WorldModelElements) do
   797→            if elem._defaultSnapshot then
   798→                for k, v in pairs(elem._defaultSnapshot) do
   799→                    elem[k] = v
   800→                end
   801→            else
   802→                elem.active = elem._defaultActive or false
   803→            end
   804→        end
   805→    end
   806→
   807→    -- 4. RESET BODYGROUPS
   808→    self:ResetBodygroups()
   809→
   810→    -- 5. APPLY each equipped attachment (respecting defaults)
   811→    print("[CUH-DBG]   ApplyAttachments step 5: iterating " .. #self.Attachments .. " slots")
   812→    for slot = 1, #self.Attachments do
   813→        local slotData = self.Attachments[slot]
   814→        if slotData then
   815→            local sel = self:GetEffectiveAttachment(slot)
   816→            print("[CUH-DBG]     slot=" .. slot .. "  sel(effective)=" .. tostring(sel)
   817→                .. "  atts count=" .. tostring(slotData.atts and #slotData.atts or 0))
   818→            if sel and sel > 0 then
   819→                local attId = slotData.atts[sel]
   820→                print("[CUH-DBG]       attId = " .. tostring(attId) .. "  (slotData.atts[" .. sel .. "])")
   821→                if attId and CustomUH and CustomUH.Attachments and CustomUH.Attachments[attId] then
   822→                    local att = CustomUH.Attachments[attId]
   823→                    print("[CUH-DBG]       att FOUND:  Name=" .. tostring(att.Name) .. "  PartClass=" .. tostring(att.PartClass) .. "  ModelPath=" .. tostring(att.ModelPath))
   824→                    print("[CUH-DBG]       att.Attach = " .. tostring(att.Attach) .. "  att.WeaponTable = " .. tostring(att.WeaponTable))
   825→                    print("[CUH-DBG]       ViewModelElements['" .. tostring(att.PartClass) .. "'] present: "
   826→                        .. tostring(self.ViewModelElements and self.ViewModelElements[att.PartClass] ~= nil))
   827→                    if self.ViewModelElements and self.ViewModelElements[att.PartClass] then
   828→                        print("[CUH-DBG]         current elem.model = " .. tostring(self.ViewModelElements[att.PartClass].model))
   829→                    end
   830→
   831→                    -- Apply WeaponTable stats
   832→                    if att.WeaponTable then
   833→                        for path, value in pairs(att.WeaponTable) do
   834→                            if path ~= "VElements" and path ~= "WElements"
   835→                               and path ~= "Bodygroups_V" and path ~= "Bodygroups_W"
   836→                               and path ~= "ViewModelBoneMods" and path ~= "WorldModelBoneMods"
   837→                               and path ~= "Animations" then
   838→                                -- Handle nested Primary/Secondary tables:
   839→                                -- iterate sub-keys and apply each as
   840→                                -- "Primary.Spread", "Primary.Delay", etc.
   841→                                -- Otherwise SetStat("Primary", table) would
   842→                                -- REPLACE the entire Primary table, losing
   843→                                -- Damage / ClipSize / Ammo / etc.
   844→                                if (path == "Primary" or path == "Secondary") and istable(value) then
   845→                                    for subKey, subVal in pairs(value) do
   846→                                        local fullPath = path .. "." .. subKey
   847→                                        if isfunction(subVal) then
   848→                                            local currentVal = self:GetStat(fullPath)
   849→                                            -- Defensive: if the weapon doesn't define this stat
   850→                                            -- (currentVal is nil) we cannot meaningfully transform
   851→                                            -- it — calling subVal(self, nil) would crash inside
   852→                                            -- the attachment's `val * 0.8` etc. and abort the
   853→                                            -- ENTIRE ApplyAttachments, which prevents the model
   854→                                            -- swap and bodygroup changes from running too
   855→                                            -- (visible as "attachments don't apply at all").
   856→                                            -- Wrap in pcall as a final safety net so a buggy
   857→                                            -- attachment function never breaks the whole pipeline.
   858→                                            if currentVal ~= nil then
   859→                                                local ok, newVal, shouldSet = pcall(subVal, self, currentVal)
   860→                                                if ok and shouldSet ~= false and newVal ~= nil then
   861→                                                    self:SetStat(fullPath, newVal)
   862→                                                elseif not ok then
   863→                                                    print("[CUH-DBG]   WARNING: attachment stat function crashed for "
   864→                                                        .. tostring(fullPath) .. ": " .. tostring(newVal))
   865→                                                end
   866→                                            else
   867→                                                print("[CUH-DBG]   SKIP: stat " .. tostring(fullPath)
   868→                                                    .. " not defined on weapon — function transform ignored")
   869→                                            end
   870→                                        else
   871→                                            self:SetStat(fullPath, subVal)
   872→                                        end
   873→                                    end
   874→                                elseif isfunction(value) then
   875→                                    local currentVal = self:GetStat(path)
   876→                                    if currentVal ~= nil then
   877→                                        local ok, newVal, shouldSet = pcall(value, self, currentVal)
   878→                                        if ok and shouldSet ~= false and newVal ~= nil then
   879→                                            self:SetStat(path, newVal)
   880→                                        elseif not ok then
   881→                                            print("[CUH-DBG]   WARNING: attachment stat function crashed for "
   882→                                                .. tostring(path) .. ": " .. tostring(newVal))
   883→                                        end
   884→                                    else
   885→                                        print("[CUH-DBG]   SKIP: stat " .. tostring(path)
   886→                                            .. " not defined on weapon — function transform ignored")
   887→                                    end
   888→                                else
   889→                                    self:SetStat(path, value)
   890→                                end
   891→                            end
   892→                        end
   893→
   894→                        -- Bodygroups_V
   895→                        if att.WeaponTable.Bodygroups_V then
   896→                            self.Bodygroups_V = self.Bodygroups_V or {}
   897→                            for bg, val in pairs(att.WeaponTable.Bodygroups_V) do
   898→                                self.Bodygroups_V[bg] = val
   899→                            end
   900→                        end
   901→                        -- Bodygroups_W
   902→                        if att.WeaponTable.Bodygroups_W then
   903→                            self.Bodygroups_W = self.Bodygroups_W or {}
   904→                            for bg, val in pairs(att.WeaponTable.Bodygroups_W) do
   905→                                self.Bodygroups_W[bg] = val
   906→                            end
   907→                        end
   908→                    end
   909→
   910→                    -- VElement overrides
   911→                    if att.WeaponTable and att.WeaponTable.VElements then
   912→                        if not self.ViewModelElements then self.ViewModelElements = {} end
   913→                        for name, override in pairs(att.WeaponTable.VElements) do
   914→                            if self.ViewModelElements[name] then
   915→                                for k, v in pairs(override) do
   916→                                    self.ViewModelElements[name][k] = v
   917→                                end
   918→                            else
   919→                                self.ViewModelElements[name] = table.Copy(override)
   920→                            end
   921→                        end
   922→                    end
   923→
   924→                    -- WElement overrides
   925→                    if att.WeaponTable and att.WeaponTable.WElements then
   926→                        if not self.WorldModelElements then self.WorldModelElements = {} end
   927→                        for name, override in pairs(att.WeaponTable.WElements) do
   928→                            if self.WorldModelElements[name] then
   929→                                for k, v in pairs(override) do
   930→                                    self.WorldModelElements[name][k] = v
   931→                                end
   932→                            else
   933→                                self.WorldModelElements[name] = table.Copy(override)
   934→                            end
   935→                        end
   936→                    end
   937→
   938→                    -- Animations overrides — merges per-key into self.Animations.
   939→                    -- Values can be either:
   940→                    --   * a string (sequence name or ACT_VM_* name) -> direct assignment
   941→                    --   * a function(wep, val) -> called, result stored
   942→                    -- Example:
   943→                    --   ATTACHMENT.WeaponTable = {
   944→                    --       ["Animations"] = {
   945→                    --           ["reload"]       = "reload_ext02",
   946→                    --           ["reload_empty"] = "reload_empty_ext02",
   947→                    --       },
   948→                    --   }
   949→                    if att.WeaponTable and att.WeaponTable.Animations then
   950→                        if not self.Animations then self.Animations = {} end
   951→                        for animKey, animVal in pairs(att.WeaponTable.Animations) do
   952→                            if isfunction(animVal) then
   953→                                local cur = self.Animations[animKey]
   954→                                local ok, newVal = pcall(animVal, self, cur)
   955→                                if ok and newVal ~= nil then
   956→                                    self.Animations[animKey] = newVal
   957→                                elseif not ok then
   958→                                    ErrorNoHalt("[CUH] Attachment " .. tostring(attId)
   959→                                        .. " Animations[" .. tostring(animKey)
   960→                                        .. "] function crashed: " .. tostring(newVal) .. "\n")
   961→                                end
   962→                            else
   963→                                self.Animations[animKey] = animVal
   964→                            end
   965→                        end
   966→                    end
   967→
   968→                    -- AnimSounds overrides — same pattern as Animations.
   969→                    if att.WeaponTable and att.WeaponTable.AnimSounds then
   970→                        if not self.AnimSounds then self.AnimSounds = {} end
   971→                        for key, val in pairs(att.WeaponTable.AnimSounds) do
   972→                            if isfunction(val) then
   973→                                local cur = self.AnimSounds[key]
   974→                                local ok, newVal = pcall(val, self, cur)
   975→                                if ok and newVal ~= nil then
   976→                                    self.AnimSounds[key] = newVal
   977→                                elseif not ok then
   978→                                    ErrorNoHalt("[CUH] Attachment " .. tostring(attId)
   979→                                        .. " AnimSounds[" .. tostring(key)
   980→                                        .. "] function crashed: " .. tostring(newVal) .. "\n")
   981→                                end
   982→                            else
   983→                                self.AnimSounds[key] = val
   984→                            end
   985→                        end
   986→                    end
   987→
   988→                    -- Call attachment's Attach function for side effects
   989→                    -- (e.g. model swap via wep.ViewModelElements[PartClass].model = newModel)
   990→                    -- Wrapped in pcall so a buggy attachment never breaks the
   991→                    -- entire ApplyAttachments pipeline (which would prevent
   992→                    -- ALL subsequent slots from being applied).
   993→                    if att.Attach then
   994→                        print("[CUH-DBG]       calling att:Attach(self) ...")
   995→                        local ok, err = pcall(att.Attach, att, self)
   996→                        if not ok then
   997→                            print("[CUH-DBG]       ERROR in att:Attach: " .. tostring(err))
   998→                            ErrorNoHalt("[CUH] Attachment " .. tostring(attId) .. " Attach() failed: " .. tostring(err) .. "\n")
   999→                        else
  1000→                            print("[CUH-DBG]       att:Attach returned. elem.model now = "
  1001→                                .. tostring(self.ViewModelElements and self.ViewModelElements[att.PartClass] and self.ViewModelElements[att.PartClass].model))
  1002→                        end
  1003→                    else
  1004→                        print("[CUH-DBG]       WARNING: att.Attach is nil — model swap will NOT happen")
  1005→                    end
  1006→                else
  1007→                    print("[CUH-DBG]       NOT FOUND in CustomUH.Attachments  attId=" .. tostring(attId))
  1008→                end
  1009→            else
  1010→                print("[CUH-DBG]       sel is nil or 0 — skipping")
  1011→            end
  1012→        else
  1013→            print("[CUH-DBG]     slot=" .. slot .. "  slotData is nil")
  1014→        end
  1015→    end
  1016→
  1017→    -- 6. Re-init VElements and WElements (in case models were swapped).
  1018→    --    Attach() may already have done CleanupVElements + InitVElements
  1019→    --    for its own part, but we re-run all four so EVERY element picks
  1020→    --    up the new model/bodygroup/skin/material fields. This is what
  1021→    --    actually rebuilds the ClientsideModels that DrawVElements and
  1022→    --    DrawWElements render each frame.
  1023→    print("[CUH-DBG]   ApplyAttachments step 6: CleanupVElements + InitVElements (rebuild ClientsideModels)")
  1024→    if self.ViewModelElements then
  1025→        for name, elem in pairs(self.ViewModelElements) do
  1026→            print("[CUH-DBG]     VElement['" .. tostring(name) .. "']  model=" .. tostring(elem.model)
  1027→                .. "  _csModel=" .. tostring(elem._csModel) .. "  active=" .. tostring(elem.active))
  1028→        end
  1029→    end
  1030→    if self.CleanupVElements then self:CleanupVElements() end
  1031→    if self.InitVElements then self:InitVElements() end
  1032→    if self.CleanupWElements then self:CleanupWElements() end
  1033→    if self.InitWElements then self:InitWElements() end
  1034→    print("[CUH-DBG]   post-rebuild VElement barrel _csModel = "
  1035→        .. tostring(self.ViewModelElements and self.ViewModelElements["barrel"] and self.ViewModelElements["barrel"]._csModel))
  1036→    print("[CUH-DBG] <<< ApplyAttachments returning")
  1037→end

  1039→-- ============================================================
  1040→-- SET ATTACHMENT — select an attachment in a slot
  1041→-- ============================================================
  1042→
  1043→function SWEP:SetAttachment(slot, index)
  1044→    print("[CUH-DBG] >>> SetAttachment CALLED  slot=" .. tostring(slot) .. " index=" .. tostring(index)
  1045→        .. " realm=" .. (SERVER and "SERVER" or "CLIENT") .. " wep=" .. tostring(self)
  1046→        .. " class=" .. tostring(self.GetClass and self:GetClass() or "?"))
  1047→
  1048→    if not self.Attachments or not self.Attachments[slot] then
  1049→        print("[CUH-DBG]   ABORT: self.Attachments missing or no slot " .. tostring(slot))
  1050→        return
  1051→    end
  1052→
  1053→    -- Validate index
  1054→    if index > 0 and (not self.Attachments[slot].atts or index > #self.Attachments[slot].atts) then
  1055→        print("[CUH-DBG]   ABORT: index out of range (atts count=" .. tostring(self.Attachments[slot].atts and #self.Attachments[slot].atts) .. ")")
  1056→        return
  1057→    end
  1058→
  1059→    -- forceDefault check: if player tries to select "None" (0)
  1060→    -- and forceDefault is on, snap back to default instead
  1061→    if index == 0 and self.Attachments[slot].forceDefault
  1062→        and self.Attachments[slot].default and self.Attachments[slot].default > 0 then
  1063→        index = self.Attachments[slot].default
  1064→        print("[CUH-DBG]   forceDefault: snapped 0 -> " .. tostring(index))
  1065→    end
  1066→
  1067→    self.Attachments[slot].sel = index
  1068→    print("[CUH-DBG]   self.Attachments[" .. slot .. "].sel = " .. tostring(index)
  1069→        .. "  atts[index]=" .. tostring(self.Attachments[slot].atts and self.Attachments[slot].atts[index]))
  1070→    print("[CUH-DBG]   CustomUH.Attachments has " .. tostring(CustomUH and CustomUH.Attachments and table.Count(CustomUH.Attachments) or "nil/empty") .. " entries")
  1071→    print("[CUH-DBG]   calling self:ApplyAttachments() ...")
  1072→    self:ApplyAttachments()
  1073→    print("[CUH-DBG] <<< SetAttachment returning (realm=" .. (SERVER and "SERVER" or "CLIENT") .. ")")
  1074→
  1075→    -- Network to server
  1076→    if CLIENT then
  1077→        print("[CUH-DBG]   CLIENT sending net CUH2_AttSelect (slot=" .. slot .. " index=" .. index .. ")")
  1078→        net.Start("CUH2_AttSelect")
  1079→            net.WriteEntity(self)
  1080→            net.WriteUInt(slot, 8)
  1081→            net.WriteUInt(index, 8)
  1082→        net.SendToServer()
  1083→    end
  1084→
  1085→    -- Save on server + broadcast to all clients
  1086→    -- When SetAttachment is called directly on the server (e.g. via
  1087→    -- the cuh_set console command, or via the CUH2_AttSelect net
  1088→    -- receiver), we MUST broadcast CUH2_AttSync to all clients so
  1089→    -- they apply the same attachment locally. Without this, the
  1090→    -- server's copy of the weapon has the swapped model string,
  1091→    -- but the clients' copies don't — so no visual change appears.
  1092→    if SERVER then
  1093→        if CustomUH.SaveAttachments then
  1094→            CustomUH.SaveAttachments(self, self.Owner)
  1095→        end
  1096→        print("[CUH-DBG]   SERVER broadcasting CUH2_AttSync (slot=" .. slot .. " index=" .. index .. ")")
  1097→        net.Start("CUH2_AttSync")
  1098→            net.WriteEntity(self)
  1099→            net.WriteUInt(slot, 8)
  1100→            net.WriteUInt(index, 8)
  1101→        net.Broadcast()
  1102→    end
  1103→end

  1118→-- Plays the melee animation, schedules the hit trace, and
  1119→-- locks controls for the duration.
  1120→function SWEP:MeleeAttack()
  1121→    local ct = CurTime()
  1122→    if not (game.SinglePlayer() or IsFirstTimePredicted()) then return end
  1123→
  1124→    self:SetUHBool("Zooming", false)
  1125→    if self:GetUHBool("Running") then self:SetUHBool("Running", false) end
  1126→
  1127→    if self.MeleeInterruptReload ~= false and self:GetUHBool("Reloading") then
  1128→        self:SetUHBool("Reloading", false)
  1129→        self:SetNWFloat("ReloadTime", 0)
  1130→        self:SetNWFloat("ReloadEndTime", 0)
  1131→        if timer.Exists("UHReload_"..self.Owner:SteamID()) then
  1132→            timer.Remove("UHReload_"..self.Owner:SteamID())
  1133→        end
  1134→    end
  1135→
  1136→    self._meleeActive = true
  1137→    self:ClearAnimSounds()
  1138→    self:EasySendWeaponAnim("melee", ACT_VM_MELEE)
  1139→
  1140→    local vm = IsValid(self.Owner) and self.Owner:GetViewModel() or nil
  1141→    local animDuration = IsValid(vm) and vm:SequenceDuration() or 0.5
  1142→    self._meleeHitTime = ct + (self.MeleeHitDelay or 0.15)
  1143→    self._meleeHitDone = false
  1144→    self._meleeEndTime = ct + animDuration
  1145→    self._nextMelee = ct + (self.MeleeDelay or 0.6)
  1146→    self:SetNextPrimaryFire(ct + animDuration)
  1147→    self:SetNextSecondaryFire(ct + animDuration)
  1148→    self.NextReload = ct + animDuration
  1149→
  1150→    local seqIdx = self.Owner:SelectWeightedSequence(ACT_GMOD_GESTURE_MELEE_SHOVE_2HAND)
  1151→    if seqIdx and seqIdx >= 0 then
  1152→        self.Owner:AddVCDSequenceToGestureSlot(GESTURE_SLOT_ATTACK_AND_RELOAD, seqIdx, 0, true)
  1153→    end
  1154→
  1155→    local snd = self.MeleeSound
  1156→    if istable(snd) then snd = snd[math.random(1, #snd)] end
  1157→    if snd and snd ~= "" then
  1158→        self:EmitSound(snd, 75, 100, 1, CHAN_USER_BASE)
  1159→    end
  1160→end

  1162→function SWEP:DoMeleeTrace()
  1163→    local ply = self.Owner
  1164→    if not IsValid(ply) then return end
  1165→    local pos = ply:GetShootPos()
  1166→    local aim = ply:GetAimVector()
  1167→    local range = self.MeleeRange or 64
  1168→    local tr = util.TraceHull({
  1169→        start = pos, endpos = pos + aim * range,
  1170→        filter = ply, mask = MASK_SHOT_HULL,
  1171→        mins = Vector(-10,-10,-10), maxs = Vector(10,10,10),
  1172→    })
  1173→    if tr.Hit then
  1174→        if IsValid(tr.Entity) and SERVER then
  1175→            local dmg = DamageInfo()
  1176→            dmg:SetDamage(self.MeleeDamage or 50)
  1177→            dmg:SetAttacker(ply)
  1178→            dmg:SetInflictor(self)
  1179→            dmg:SetDamageForce(aim * (self.MeleeForce or 300))
  1180→            dmg:SetDamagePosition(tr.HitPos)
  1181→            dmg:SetDamageType(DMG_CLUB)
  1182→            tr.Entity:TakeDamageInfo(dmg)
  1183→        end
  1184→        if SERVER then util.ScreenShake(tr.HitPos, 3, 0.1, 0.3, 32) end
  1185→        local hitSnd = self.MeleeHitSound
  1186→        if istable(hitSnd) then hitSnd = hitSnd[math.random(1, #hitSnd)] end
  1187→        if hitSnd and hitSnd ~= "" then
  1188→            self:EmitSound(hitSnd, 75, 100, 1, CHAN_USER_BASE)
  1189→        end
  1190→    else
  1191→        if self.MeleeMissSound and self.MeleeMissSound ~= "" then
  1192→            self:EmitSound(self.MeleeMissSound, 65, 100, 1, CHAN_USER_BASE)
  1193→        end
  1194→    end
  1195→    ply:ViewPunch(self.MeleeViewPunch or Angle(-3,0,0))
  1196→end
